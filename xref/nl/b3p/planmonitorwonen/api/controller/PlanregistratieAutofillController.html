7   package nl.b3p.planmonitorwonen.api.controller;
8   
9   import static org.springframework.http.HttpStatus.BAD_REQUEST;
10  
11  import java.util.List;
12  import java.util.Map;
13  import nl.b3p.planmonitorwonen.api.service.PlanregistratieAutofillService;
14  import org.locationtech.jts.geom.Geometry;
15  import org.locationtech.jts.geom.GeometryFactory;
16  import org.locationtech.jts.geom.PrecisionModel;
17  import org.locationtech.jts.io.ParseException;
18  import org.locationtech.jts.io.WKTReader;
19  import org.springframework.context.annotation.Profile;
20  import org.springframework.jdbc.core.simple.JdbcClient;
21  import org.springframework.web.bind.annotation.PostMapping;
22  import org.springframework.web.bind.annotation.RequestBody;
23  import org.springframework.web.bind.annotation.RestController;
24  import org.springframework.web.server.ResponseStatusException;
25  
26  @RestController
27  @Profile("!test")
28  public class PlanregistratieAutofillController {
29    private final JdbcClient jdbcClient;
30    private final PlanregistratieAutofillService autofillService;
31  
32    public PlanregistratieAutofillController(JdbcClient jdbcClient, PlanregistratieAutofillService autofillService) {
33      this.jdbcClient = jdbcClient;
34      this.autofillService = autofillService;
35    }
36  
37    // See https://www.w3.org/TR/sdw-bp/#applicability-formatVbp, use text/plain content type for WKT
38    // in request body
39    @PostMapping(
40        path = "${planmonitor-wonen-api.base-path}/planregistratie/autofill-by-geometry",
41        consumes = "text/plain")
42    public Map<String, Object> autofill(@RequestBody String wkt) {
43      Geometry geometry;
44      try {
45        GeometryFactory geometryFactory = new GeometryFactory(new PrecisionModel(), 28992);
46        geometry = new WKTReader(geometryFactory).read(wkt);
47      } catch (ParseException e) {
48        throw new ResponseStatusException(BAD_REQUEST, "WKT parsing error");
49      }
50  
51      List<String> gemeentes = this.jdbcClient
52          .sql("""
53  select g.naam
54  from gemeente g
55  where st_intersects(g.geometry, st_geomfromtext(?, 28992))
56  order by st_area(st_intersection(g.geometry, st_geomfromtext(?, 28992))) desc""")
57          .param(wkt)
58          .param(wkt)
59          .query(String.class)
60          .list();
61  
62      return Map.of(
63          "gemeentes", gemeentes,
64          "provincies", autofillService.getProvincies(geometry),
65          "regios", autofillService.getRegios(geometry),
66          "woonmilieus", autofillService.getWoonmilieus(geometry));
67    }
68  }
