38  @RestController
39  @RequestMapping(path = "${planmonitor-wonen-api.base-path}/{version}", version = MAJOR_VERSION)
40  @Profile("!test")
41  @Tag(name = WEBAPP_TAG, description = WEBAPP_TAG_DESCRIPTION)
42  public class PlanregistratieAutofillController {
43    private final JdbcClient jdbcClient;
44    private final PlanregistratieAutofillService autofillService;
45    private final PlanmonitorAuthenticationService authenticationService;
46  
47    public PlanregistratieAutofillController(
48        JdbcClient jdbcClient,
49        PlanregistratieAutofillService autofillService,
50        PlanmonitorAuthenticationService authenticationService) {
51      this.jdbcClient = jdbcClient;
52      this.autofillService = autofillService;
53      this.authenticationService = authenticationService;
54    }
55  
56    // See https://www.w3.org/TR/sdw-bp/#applicability-formatVbp, use text/plain content type for WKT
57    // in request body
58    @PostMapping(path = "/planregistratie/autofill-by-geometry", consumes = "text/plain")
59    @Operation(
60        summary = "Gemeentes, provincies, regio's en woonmilieus die een geometrie snijden",
61        description = "De request body is een geometrie als WKT in RD New (EPSG:28992). De gemeentes zijn"
62            + " gesorteerd op de grootte van de overlap, de grootste eerst. Voor het vooraf invullen van het"
63            + " formulier van de webapplicatie, een client heeft dit niet nodig: de server vult `provincie`, `regio`"
64            + " en `beoogdWoonmilieuAbf13` bij het opslaan zelf in.")
65    @ApiResponse(responseCode = "200", description = "De gevonden waarden per veld")
66    @ApiResponse(responseCode = "400", description = "Ongeldige WKT")
67    @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
68    @ApiResponse(responseCode = "403", ref = FORBIDDEN)
69    public Map<String, Object> autofill(
70        @io.swagger.v3.oas.annotations.parameters.RequestBody(
71                content =
72                    @Content(
73                        mediaType = "text/plain",
74                        schema =
75                            @Schema(
76                                example =
77                                    "POLYGON ((31400 391900, 31600 391900, 31600 392100, 31400 392100, 31400 391900))")))
78            @RequestBody
79            String wkt) {
80      // Only for users who can edit plans, this runs spatial queries on an arbitrary geometry
81      authenticationService.requirePlanmonitorUser();
82  
83      Geometry geometry;
84      try {
85        GeometryFactory geometryFactory = new GeometryFactory(new PrecisionModel(), 28992);
86        geometry = new WKTReader(geometryFactory).read(wkt);
87      } catch (ParseException e) {
88        throw new ResponseStatusException(BAD_REQUEST, "WKT parsing error");
89      }
90  
91      List<String> gemeentes = this.jdbcClient
92          .sql("""
93  select g.naam
94  from gemeente g
95  where st_intersects(g.geometry, st_geomfromtext(?, 28992))
96  order by st_area(st_intersection(g.geometry, st_geomfromtext(?, 28992))) desc""")
97          .param(wkt)
98          .param(wkt)
99          .query(String.class)
100         .list();
101 
102     return Map.of(
103         "gemeentes", gemeentes,
104         "provincies", autofillService.getProvincies(geometry),
105         "regios", autofillService.getRegios(geometry),
106         "woonmilieus", autofillService.getWoonmilieus(geometry));
107   }
108 }
