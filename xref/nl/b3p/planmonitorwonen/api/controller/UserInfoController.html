29  /**
30   * Shows the logged-in user as seen by this API, to debug sharing the Tailormap login session. Same as the /user
31   * endpoint of other Tailormap backends, plus the result of
32   * {@link PlanmonitorAuthenticationService#requirePlanmonitorUser()}. Requires a login but no gemeente or provincie.
33   */
34  @RestController
35  @RequestMapping(path = "${planmonitor-wonen-api.base-path}/{version}", version = MAJOR_VERSION)
36  @Profile("!test")
37  @Tag(name = "Gebruiker")
38  public class UserInfoController {
39    private final PlanmonitorAuthenticationService authenticationService;
40  
41    public UserInfoController(PlanmonitorAuthenticationService authenticationService) {
42      this.authenticationService = authenticationService;
43    }
44  
45    @GetMapping(path = "/user", produces = MediaType.APPLICATION_JSON_VALUE)
46    @Operation(
47        summary = "De ingelogde gebruiker",
48        description =
49            "Gebruikersnaam, groepen (`authorities`) en eigenschappen van de ingelogde gebruiker. Voor een"
50                + " gebruiker met autorisatie voor Planmonitor Wonen ook `isProvincie` en `gemeentes` (de gemeentes"
51                + " waarvoor de gebruiker planregistraties mag aanleveren), anders de reden in `planmonitorWonenError`.")
52    @ApiResponse(responseCode = "200", description = "De ingelogde gebruiker")
53    @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
54    public Map<String, Object> user() {
55      Authentication authentication = authenticationService.requireAuthenticated();
56  
57      Map<String, Object> response = new HashMap<>();
58      response.put("username", authentication.getName());
59      response.put(
60          "authorities",
61          authentication.getAuthorities().stream()
62              .map(GrantedAuthority::getAuthority)
63              .toList());
64  
65      if (authentication.getPrincipal() instanceof TailormapUserDetails userDetails) {
66        response.put("organisation", userDetails.getOrganisation());
67        response.put(
68            "additionalProperties",
69            userDetails.getAdditionalProperties().stream()
70                .filter(TailormapAdditionalProperty::isPublic)
71                .toList());
72        response.put(
73            "additionalGroupProperties",
74            userDetails.getAdditionalGroupProperties().stream()
75                .filter(TailormapAdditionalProperty::isPublic)
76                .toList());
77      }
78  
79      try {
80        PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
81            authenticationService.requirePlanmonitorUser();
82        response.put("isProvincie", auth.isProvincie());
83        response.put("gemeentes", auth.gemeentes());
84      } catch (ResponseStatusException e) {
85        response.put("planmonitorWonenError", e.getReason());
86      }
87      return response;
88    }
89  }
