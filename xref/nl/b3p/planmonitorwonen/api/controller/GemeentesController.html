7   package nl.b3p.planmonitorwonen.api.controller;
8   
9   import static nl.b3p.planmonitorwonen.api.configuration.ApiVersionConfiguration.MAJOR_VERSION;
10  import static nl.b3p.planmonitorwonen.api.configuration.OpenApiConfiguration.UNAUTHORIZED;
11  import static nl.b3p.planmonitorwonen.api.configuration.OpenApiConfiguration.WEBAPP_TAG;
12  import static nl.b3p.planmonitorwonen.api.configuration.OpenApiConfiguration.WEBAPP_TAG_DESCRIPTION;
13  
14  import io.swagger.v3.oas.annotations.Operation;
15  import io.swagger.v3.oas.annotations.Parameter;
16  import io.swagger.v3.oas.annotations.responses.ApiResponse;
17  import io.swagger.v3.oas.annotations.tags.Tag;
18  import java.util.List;
19  import nl.b3p.planmonitorwonen.api.model.Gemeente;
20  import nl.b3p.planmonitorwonen.api.security.PlanmonitorAuthenticationService;
21  import org.springframework.context.annotation.Profile;
22  import org.springframework.http.ResponseEntity;
23  import org.springframework.jdbc.core.simple.JdbcClient;
24  import org.springframework.web.bind.annotation.GetMapping;
25  import org.springframework.web.bind.annotation.RequestMapping;
26  import org.springframework.web.bind.annotation.RequestParam;
27  import org.springframework.web.bind.annotation.RestController;
28  
29  @RestController
30  @RequestMapping(path = "${planmonitor-wonen-api.base-path}/{version}", version = MAJOR_VERSION)
31  @Profile("!test")
32  @Tag(name = WEBAPP_TAG, description = WEBAPP_TAG_DESCRIPTION)
33  public class GemeentesController {
34    private final JdbcClient jdbcClient;
35    private final PlanmonitorAuthenticationService authenticationService;
36  
37    public GemeentesController(JdbcClient jdbcClient, PlanmonitorAuthenticationService authenticationService) {
38      this.jdbcClient = jdbcClient;
39      this.authenticationService = authenticationService;
40    }
41  
42    @GetMapping(path = "/gemeentes")
43    @Operation(
44        summary = "Alle gemeentes, gesorteerd op naam",
45        description = "Voor de keuzelijsten van de webapplicatie, een client heeft dit niet nodig.")
46    @ApiResponse(responseCode = "200", description = "De gemeentes")
47    @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
48    public ResponseEntity<List<Gemeente>> planregistraties(
49        @Parameter(description = "Alleen de gemeentes van deze provincie", example = "Zeeland")
50            @RequestParam(required = false)
51            String provincie) {
52      // Any logged-in user, a Tailormap admin needs the gemeentes for the group properties
53      authenticationService.requireAuthenticated();
54  
55      JdbcClient.StatementSpec statementSpec;
56      if (provincie == null) {
57        statementSpec = jdbcClient.sql("select identificatie, naam, provincie from gemeente order by naam");
58      } else {
59        statementSpec = jdbcClient
60            .sql("select identificatie, naam, provincie from gemeente where provincie = ? order by naam")
61            .param(provincie);
62      }
63      List<Gemeente> gemeentes = statementSpec.query(Gemeente.class).list();
64      return ResponseEntity.ok(gemeentes);
65    }
66  }
