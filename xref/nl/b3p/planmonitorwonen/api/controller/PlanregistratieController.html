9   import static org.springframework.http.HttpStatus.BAD_REQUEST;
10  import static org.springframework.http.HttpStatus.FORBIDDEN;
11  import static org.springframework.http.HttpStatus.NOT_FOUND;
12  import static org.springframework.http.HttpStatus.OK;
13  
14  import java.lang.invoke.MethodHandles;
15  import java.util.List;
16  import java.util.UUID;
17  import nl.b3p.planmonitorwonen.api.PlanmonitorWonenDatabaseService;
18  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
19  import nl.b3p.planmonitorwonen.api.security.PlanmonitorAuthenticationService;
20  import nl.b3p.planmonitorwonen.api.service.PlanregistratieCompletionService;
21  import nl.b3p.planmonitorwonen.api.validation.PlanregistratieValidationException;

32  import org.springframework.web.bind.annotation.PutMapping;
33  import org.springframework.web.bind.annotation.RequestBody;
34  import org.springframework.web.bind.annotation.RequestParam;
35  import org.springframework.web.bind.annotation.RestController;
36  import org.springframework.web.server.ResponseStatusException;
37  
38  @RestController
39  @Profile("!test")
40  public class PlanregistratieController {
41    private static final Logger logger =
42        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
43  
44    private final PlanmonitorAuthenticationService authenticationService;
45    private final PlanmonitorWonenDatabaseService pmwDb;
46    private final PlanregistratieCompletionService completionService;
47  
48    public PlanregistratieController(
49        PlanmonitorAuthenticationService authenticationService,
50        PlanmonitorWonenDatabaseService planmonitorWonenDatabaseService,
51        PlanregistratieCompletionService completionService) {
52      this.authenticationService = authenticationService;
53      this.pmwDb = planmonitorWonenDatabaseService;
54      this.completionService = completionService;
55    }
56  
57    @GetMapping(path = "${planmonitor-wonen-api.base-path}/planregistraties")
58    public List<Planregistratie> planregistraties(@RequestParam(required = false) boolean details) {
59      PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
60          authenticationService.getFromSecurityContext();
61      if (auth.isProvincie()) {
62        return pmwDb.getPlanregistratiesForProvincie(details);
63      } else {
64        return pmwDb.getPlanregistratiesForGemeentes(auth.gemeentes(), details);
65      }
66    }
67  
68    @GetMapping(path = "${planmonitor-wonen-api.base-path}/planregistratie/{id}")
69    public Planregistratie planregistratie(@PathVariable("id") UUID id) {
70      PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
71          authenticationService.getFromSecurityContext();
72  
73      Planregistratie planregistratie =
74          pmwDb.getPlanregistratie(id.toString()).orElseThrow(() -> new ResponseStatusException(NOT_FOUND));
75  
76      if (!auth.isProvincie() && !auth.gemeentes().contains(planregistratie.getGemeente())) {
77        logger.warn(
78            "Gemeente user \"{}\" with authorization for gemeentes {} tried to access plan id {} of gemeente {}, denied",
79            auth.authentication().getName(),
80            auth.gemeentes(),
81            id,
82            planregistratie.getGemeente());
83        throw new ResponseStatusException(NOT_FOUND);
84      }
85      return planregistratie;
86    }
87  
88    @PutMapping(path = "${planmonitor-wonen-api.base-path}/planregistratie/{id}")
89    public ResponseEntity<?> put(@PathVariable("id") UUID id, @RequestBody Planregistratie planregistratie)
90        throws ParseException {
91  
92      if (planregistratie.getId() != null && !isSameUuid(id, planregistratie.getId())) {
93        throw new ResponseStatusException(BAD_REQUEST, "Id in de body komt niet overeen met het id in het pad");
94      }
95      planregistratie.setId(id.toString());
96  
97      List<ValidationError> errors = PlanregistratieValidator.validate(planregistratie);
98      if (!errors.isEmpty()) {
99        throw new PlanregistratieValidationException(errors);
100     }
101 
102     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
103         authenticationService.getFromSecurityContext();
104 
105     if (auth.isProvincie()) {
106       logger.warn(
107           "Provincie user \"{}\" tried to save plan id {}, denied",
108           auth.authentication().getName(),
109           id);
110       throw new ResponseStatusException(FORBIDDEN);
111     }
112 
113     if (!auth.gemeentes().contains(planregistratie.getGemeente())) {
114       logger.warn(
115           "Gemeente user \"{}\" with authorization for gemeentes {} tried to save plan id {}, name \"{}\" with gemeente value {}, denied",
116           auth.authentication().getName(),
117           auth.gemeentes(),
118           id,
119           planregistratie.getPlanNaam(),
120           planregistratie.getGemeente());
121       throw new ResponseStatusException(FORBIDDEN);
122     }
123 
124     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
125 
126     if (gemeente != null && !auth.isProvincie() && !auth.gemeentes().contains(gemeente)) {
127       logger.warn(
128           "Gemeente user \"{}\" with authorization for gemeentes {} tried to update plan id {}, name \"{}\" of gemeente {}, denied",
129           auth.authentication().getName(),
130           auth.gemeentes(),
131           id,
132           planregistratie.getPlanNaam(),
133           gemeente);
134       throw new ResponseStatusException(FORBIDDEN);
135     }
136 
137     completionService.complete(planregistratie);
138     pmwDb.insertPlanregistratie(planregistratie, auth.authentication().getName());
139     return ResponseEntity.status(OK).build();
140   }
141 
142   @DeleteMapping(path = "${planmonitor-wonen-api.base-path}/planregistratie/{id}")
143   public ResponseEntity<?> delete(@PathVariable("id") UUID id) {
144     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
145         authenticationService.getFromSecurityContext();
146 
147     if (auth.isProvincie()) {
148       logger.warn(
149           "Provincie user \"{}\" tried to delete plan id {}, denied",
150           auth.authentication().getName(),
151           id);
152       throw new ResponseStatusException(FORBIDDEN);
153     }
154 
155     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
156 
157     if (gemeente == null) {
158       throw new ResponseStatusException(NOT_FOUND);
159     }
160 
161     if (!auth.gemeentes().contains(gemeente)) {
162       logger.warn(
163           "Gemeente user \"{}\" with authorization for gemeentes {} tried to delete plan id {} of gemeente {}, denied",
164           auth.authentication().getName(),
165           auth.gemeentes(),
166           id,
167           gemeente);
168       throw new ResponseStatusException(FORBIDDEN);
169     }
170 
171     pmwDb.deletePlanregistratie(id.toString());
172     return ResponseEntity.status(OK).build();
173   }
174 
175   private static boolean isSameUuid(UUID id, String other) {
176     try {
177       return id.equals(UUID.fromString(other));
178     } catch (IllegalArgumentException e) {
179       return false;
180     }
181   }
