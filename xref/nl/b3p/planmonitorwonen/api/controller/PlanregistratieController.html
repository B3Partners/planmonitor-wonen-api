55    public PlanregistratieController(
56        PlanmonitorAuthenticationService authenticationService,
57        PlanmonitorWonenDatabaseService planmonitorWonenDatabaseService,
58        PlanregistratieCompletionService completionService,
59        EnumValues enumValues) {
60      this.authenticationService = authenticationService;
61      this.pmwDb = planmonitorWonenDatabaseService;
62      this.completionService = completionService;
63      this.enumValues = enumValues;
64    }

66    @GetMapping(path = "/planregistraties")
67    public List<Planregistratie> planregistraties(@RequestParam(required = false) boolean details) {
68      PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
69          authenticationService.requirePlanmonitorUser();
70      if (auth.isProvincie()) {
71        return pmwDb.getPlanregistratiesForProvincie(details);
72      } else {
73        return pmwDb.getPlanregistratiesForGemeentes(auth.gemeentes(), details);
74      }
75    }
76  
77    @GetMapping(path = "/planregistratie/{id}")
78    public Planregistratie planregistratie(@PathVariable("id") UUID id) {
79      PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
80          authenticationService.requirePlanmonitorUser();
81  
82      Planregistratie planregistratie =
83          pmwDb.getPlanregistratie(id.toString()).orElseThrow(() -> new ResponseStatusException(NOT_FOUND));
84  
85      if (!auth.isProvincie() && !auth.gemeentes().contains(planregistratie.getGemeente())) {
86        logger.warn(
87            "Gemeente user \"{}\" with authorization for gemeentes {} tried to access plan id {} of gemeente {}, denied",
88            auth.authentication().getName(),
89            auth.gemeentes(),
90            id,
91            planregistratie.getGemeente());
92        throw new ResponseStatusException(NOT_FOUND);
93      }
94      return planregistratie;
95    }

97    /**
98     * Creates or replaces a planregistratie. Returns the saved planregistratie including the fields filled in by the
99     * server, with status 201 when it was created and 200 when it was replaced.
100    */
101   @PutMapping(path = "/planregistratie/{id}")
102   public ResponseEntity<Planregistratie> put(
103       @PathVariable("id") UUID id, @RequestBody Planregistratie planregistratie) throws ParseException {
104 
105     // Authenticate before validating, so a client which is not logged in gets a 401 instead of a 400
106     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
107         authenticationService.requirePlanmonitorUser();
108 
109     if (planregistratie.getId() != null && !isSameUuid(id, planregistratie.getId())) {
110       throw new ResponseStatusException(BAD_REQUEST, "Id in de body komt niet overeen met het id in het pad");
111     }
112     planregistratie.setId(id.toString());
113 
114     List<ValidationError> errors = PlanregistratieValidator.validate(planregistratie, enumValues);
115     if (!errors.isEmpty()) {
116       logRejected(auth, planregistratie, errors);
117       throw new PlanregistratieValidationException(errors);
118     }
119 
120     if (auth.isProvincie()) {
121       logger.warn(
122           "Provincie user \"{}\" tried to save plan id {}, denied",
123           auth.authentication().getName(),
124           id);
125       throw new ResponseStatusException(FORBIDDEN);
126     }
127 
128     if (!auth.gemeentes().contains(planregistratie.getGemeente())) {
129       logger.warn(
130           "Gemeente user \"{}\" with authorization for gemeentes {} tried to save plan id {}, name \"{}\" with gemeente value {}, denied",
131           auth.authentication().getName(),
132           auth.gemeentes(),
133           id,
134           planregistratie.getPlanNaam(),
135           planregistratie.getGemeente());
136       throw new ResponseStatusException(FORBIDDEN);
137     }
138 
139     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
140 
141     if (gemeente != null && !auth.isProvincie() && !auth.gemeentes().contains(gemeente)) {
142       logger.warn(
143           "Gemeente user \"{}\" with authorization for gemeentes {} tried to update plan id {}, name \"{}\" of gemeente {}, denied",
144           auth.authentication().getName(),
145           auth.gemeentes(),
146           id,
147           planregistratie.getPlanNaam(),
148           gemeente);
149       throw new ResponseStatusException(FORBIDDEN);
150     }
151 
152     try {
153       completionService.complete(planregistratie);
154     } catch (PlanregistratieValidationException e) {
155       logRejected(auth, planregistratie, e.getErrors());
156       throw e;
157     }
158     boolean created = gemeente == null;
159     try {
160       pmwDb.insertPlanregistratie(planregistratie, auth.authentication().getName());
161     } catch (PlanNaamInUseException e) {
162       logger.info(
163           "User \"{}\" tried to save plan id {} with name \"{}\" already in use by another plan, rejected",
164           auth.authentication().getName(),
165           id,
166           planregistratie.getPlanNaam());
167       throw e;
168     }
169     Planregistratie saved = pmwDb.getPlanregistratie(id.toString())
170         .orElseThrow(() -> new IllegalStateException("Saved planregistratie " + id + " not found"));
171     return ResponseEntity.status(created ? CREATED : OK).body(saved);
172   }
173 
174   @DeleteMapping(path = "/planregistratie/{id}")
175   public ResponseEntity<?> delete(@PathVariable("id") UUID id) {
176     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
177         authenticationService.requirePlanmonitorUser();
178 
179     if (auth.isProvincie()) {
180       logger.warn(
181           "Provincie user \"{}\" tried to delete plan id {}, denied",
182           auth.authentication().getName(),
183           id);
184       throw new ResponseStatusException(FORBIDDEN);
185     }
186 
187     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
188 
189     if (gemeente == null) {
190       throw new ResponseStatusException(NOT_FOUND);
191     }
192 
193     if (!auth.gemeentes().contains(gemeente)) {
194       logger.warn(
195           "Gemeente user \"{}\" with authorization for gemeentes {} tried to delete plan id {} of gemeente {}, denied",
196           auth.authentication().getName(),
197           auth.gemeentes(),
198           id,
199           gemeente);
200       throw new ResponseStatusException(FORBIDDEN);
201     }
202 
203     pmwDb.deletePlanregistratie(id.toString());
204     return ResponseEntity.status(OK).build();
205   }

207   private static void logRejected(
208       PlanmonitorAuthenticationService.PlanmonitorAuthentication auth,
209       Planregistratie planregistratie,
210       List<ValidationError> errors) {
211     logger.info(
212         "User \"{}\" tried to save invalid plan id {}, name \"{}\", rejected with {} error(s): {}",
213         auth.authentication().getName(),
214         planregistratie.getId(),
215         planregistratie.getPlanNaam(),
216         errors.size(),
217         errors);
218   }

220   private static boolean isSameUuid(UUID id, String other) {
221     try {
222       return id.equals(UUID.fromString(other));
223     } catch (IllegalArgumentException e) {
224       return false;
225     }
