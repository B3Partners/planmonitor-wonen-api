79    @GetMapping(path = "/planregistraties")
80    @Operation(
81        summary = "Alle planregistraties die de gebruiker mag inzien",
82        description = "Voor een gemeentegebruiker de planregistraties van de gemeentes waarvoor de gebruiker"
83            + " autorisatie heeft, voor een provinciegebruiker de planregistraties van de provincie.")
84    @ApiResponse(responseCode = "200", description = "De planregistraties")
85    @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
86    @ApiResponse(responseCode = "403", ref = FORBIDDEN)
87    public List<Planregistratie> planregistraties(
88        @Parameter(description = "Ook de aantallen van de planregistraties teruggeven")
89            @RequestParam(required = false)
90            boolean details) {
91      PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
92          authenticationService.requirePlanmonitorUser();
93      if (auth.isProvincie()) {
94        return pmwDb.getPlanregistratiesForProvincie(details);
95      } else {
96        return pmwDb.getPlanregistratiesForGemeentes(auth.gemeentes(), details);
97      }
98    }

100   @GetMapping(path = "/planregistratie/{id}")
101   @Operation(summary = "Een planregistratie met de aantallen")
102   @ApiResponse(responseCode = "200", description = "De planregistratie")
103   @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
104   @ApiResponse(responseCode = "403", ref = FORBIDDEN)
105   @ApiResponse(responseCode = "404", ref = NOT_FOUND)
106   public Planregistratie planregistratie(
107       @Parameter(description = "UUID van de planregistratie") @PathVariable("id") UUID id) {
108     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
109         authenticationService.requirePlanmonitorUser();
110 
111     Planregistratie planregistratie = pmwDb.getPlanregistratie(id.toString())
112         .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND));
113 
114     if (!auth.isProvincie() && !auth.gemeentes().contains(planregistratie.getGemeente())) {
115       logger.warn(
116           "Gemeente user \"{}\" with authorization for gemeentes {} tried to access plan id {} of gemeente {}, denied",
117           auth.authentication().getName(),
118           auth.gemeentes(),
119           id,
120           planregistratie.getGemeente());
121       throw new ResponseStatusException(HttpStatus.NOT_FOUND);
122     }
123     return planregistratie;
124   }

136   @io.swagger.v3.oas.annotations.parameters.RequestBody(

140               examples = @ExampleObject(name = "Planregistratie", ref = PLANREGISTRATIE_EXAMPLE)))

145   @ApiResponse(

148       content = @Content(schema = @Schema(implementation = Planregistratie.class)))
149   @ApiResponse(
150       responseCode = "400",
151       description =
152           "Ongeldige planregistratie, met de fouten per veld in `errors`. Ook bij ongeldige JSON, een"
153               + " ongeldig UUID of een id in de body dat niet gelijk is aan het id in het pad, dan zonder `errors`",
154       content = @Content(mediaType = PROBLEM_JSON, schema = @Schema(implementation = ProblemDetailSchema.class)))
155   @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
156   @ApiResponse(
157       responseCode = "403",
158       description = "Geen toegang: de gebruiker heeft geen autorisatie voor Planmonitor Wonen of voor de gemeente"
159           + " van de planregistratie, de bestaande planregistratie met dit id is van een andere gemeente, of de"
160           + " gebruiker is een provinciegebruiker (alleen lezen)",
161       content = @Content(mediaType = PROBLEM_JSON, schema = @Schema(implementation = ProblemDetailSchema.class)))
162   @ApiResponse(
163       responseCode = "409",
164       description = "Er bestaat al een andere planregistratie met dezelfde plannaam (uniek voor alle gemeentes),"
165           + " met een fout voor `planNaam` in `errors`",
166       content = @Content(mediaType = PROBLEM_JSON, schema = @Schema(implementation = ProblemDetailSchema.class)))
167   public ResponseEntity<Planregistratie> put(
168       @Parameter(description = "UUID van de planregistratie") @PathVariable("id") UUID id,
169       @RequestBody Planregistratie planregistratie)
170       throws ParseException {
171 
172     // Authenticate before validating, so a client which is not logged in gets a 401 instead of a 400
173     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
174         authenticationService.requirePlanmonitorUser();
175 
176     if (planregistratie.getId() != null && !isSameUuid(id, planregistratie.getId())) {
177       throw new ResponseStatusException(BAD_REQUEST, "Id in de body komt niet overeen met het id in het pad");
178     }
179     planregistratie.setId(id.toString());
180 
181     List<ValidationError> errors = PlanregistratieValidator.validate(planregistratie, enumValues);
182     if (!errors.isEmpty()) {
183       logRejected(auth, planregistratie, errors);
184       throw new PlanregistratieValidationException(errors);
185     }
186 
187     if (auth.isProvincie()) {
188       logger.warn(
189           "Provincie user \"{}\" tried to save plan id {}, denied",
190           auth.authentication().getName(),
191           id);
192       throw new ResponseStatusException(HttpStatus.FORBIDDEN);
193     }
194 
195     if (!auth.gemeentes().contains(planregistratie.getGemeente())) {
196       logger.warn(
197           "Gemeente user \"{}\" with authorization for gemeentes {} tried to save plan id {}, name \"{}\" with gemeente value {}, denied",
198           auth.authentication().getName(),
199           auth.gemeentes(),
200           id,
201           planregistratie.getPlanNaam(),
202           planregistratie.getGemeente());
203       throw new ResponseStatusException(HttpStatus.FORBIDDEN);
204     }
205 
206     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
207 
208     if (gemeente != null && !auth.isProvincie() && !auth.gemeentes().contains(gemeente)) {
209       logger.warn(
210           "Gemeente user \"{}\" with authorization for gemeentes {} tried to update plan id {}, name \"{}\" of gemeente {}, denied",
211           auth.authentication().getName(),
212           auth.gemeentes(),
213           id,
214           planregistratie.getPlanNaam(),
215           gemeente);
216       throw new ResponseStatusException(HttpStatus.FORBIDDEN);
217     }
218 
219     try {
220       completionService.complete(planregistratie);
221     } catch (PlanregistratieValidationException e) {
222       logRejected(auth, planregistratie, e.getErrors());
223       throw e;
224     }
225     boolean created = gemeente == null;
226     try {
227       pmwDb.insertPlanregistratie(planregistratie, auth.authentication().getName());
228     } catch (PlanNaamInUseException e) {
229       logger.info(
230           "User \"{}\" tried to save plan id {} with name \"{}\" already in use by another plan, rejected",
231           auth.authentication().getName(),
232           id,
233           planregistratie.getPlanNaam());
234       throw e;
235     }
236     Planregistratie saved = pmwDb.getPlanregistratie(id.toString())
237         .orElseThrow(() -> new IllegalStateException("Saved planregistratie " + id + " not found"));
238     return ResponseEntity.status(created ? CREATED : OK).body(saved);
239   }

241   @DeleteMapping(path = "/planregistratie/{id}")
242   @Operation(summary = "Een planregistratie verwijderen")
243   @ApiResponse(responseCode = "200", description = "De planregistratie is verwijderd")
244   @ApiResponse(responseCode = "401", ref = UNAUTHORIZED)
245   @ApiResponse(
246       responseCode = "403",
247       description = "Geen toegang: de gebruiker heeft geen autorisatie voor Planmonitor Wonen of voor de gemeente"
248           + " van de planregistratie, of de gebruiker is een provinciegebruiker (alleen lezen)",
249       content = @Content(mediaType = PROBLEM_JSON, schema = @Schema(implementation = ProblemDetailSchema.class)))
250   @ApiResponse(
251       responseCode = "404",
252       description = "Er is geen planregistratie met dit id. Een client kan dit behandelen als al verwijderd",
253       content = @Content(mediaType = PROBLEM_JSON, schema = @Schema(implementation = ProblemDetailSchema.class)))
254   public ResponseEntity<Void> delete(
255       @Parameter(description = "UUID van de planregistratie") @PathVariable("id") UUID id) {
256     PlanmonitorAuthenticationService.PlanmonitorAuthentication auth =
257         authenticationService.requirePlanmonitorUser();
258 
259     if (auth.isProvincie()) {
260       logger.warn(
261           "Provincie user \"{}\" tried to delete plan id {}, denied",
262           auth.authentication().getName(),
263           id);
264       throw new ResponseStatusException(HttpStatus.FORBIDDEN);
265     }
266 
267     String gemeente = pmwDb.getPlanregistratieGemeente(id.toString());
268 
269     if (gemeente == null) {
270       throw new ResponseStatusException(HttpStatus.NOT_FOUND);
271     }
272 
273     if (!auth.gemeentes().contains(gemeente)) {
274       logger.warn(
275           "Gemeente user \"{}\" with authorization for gemeentes {} tried to delete plan id {} of gemeente {}, denied",
276           auth.authentication().getName(),
277           auth.gemeentes(),
278           id,
279           gemeente);
280       throw new ResponseStatusException(HttpStatus.FORBIDDEN);
281     }
282 
283     pmwDb.deletePlanregistratie(id.toString());
284     return ResponseEntity.status(OK).build();
285   }

287   private static void logRejected(
288       PlanmonitorAuthenticationService.PlanmonitorAuthentication auth,
289       Planregistratie planregistratie,
290       List<ValidationError> errors) {
291     logger.info(
292         "User \"{}\" tried to save invalid plan id {}, name \"{}\", rejected with {} error(s): {}",
293         auth.authentication().getName(),
294         planregistratie.getId(),
295         planregistratie.getPlanNaam(),
296         errors.size(),
297         errors);
298   }

300   private static boolean isSameUuid(UUID id, String other) {
301     try {
302       return id.equals(UUID.fromString(other));
303     } catch (IllegalArgumentException e) {
304       return false;
305     }
