7   package nl.b3p.planmonitorwonen.api.model;
8   
9   import io.swagger.v3.oas.annotations.media.Schema;
10  import java.util.Objects;
11  import java.util.SortedMap;
12  import java.util.TreeMap;
13  import nl.b3p.planmonitorwonen.api.validation.AantallenValidator;
14  
15  /**
16   * Number of dwellings for a plancategorie. A year left out of perJaar is not filled in, a year with 0 is filled in as
17   * zero. For the totals both count as zero.
18   *
19   * @param gepland total number of planned dwellings, null is the same as 0
20   * @param gerealiseerd number of dwellings already realised, null is the same as 0
21   * @param perJaar planned number of dwellings per year of delivery (fasering), years with a null value are left out
22   */
23  @Schema(
24      description = "Aantal woningen van een categorie. In een response zijn `gepland` en `gerealiseerd` altijd"
25          + " ingevuld en `perJaar` bevat alleen ingevulde jaren")
26  public record Aantal(
27      @Schema(
28          description = "Totaal aantal geplande woningen, weglaten of `null` telt als 0",
29          minimum = "0",
30          example = "20")
31      Integer gepland,
32  
33      @Schema(
34          description =
35              "Aantal al gerealiseerde woningen, niet meer dan `gepland`. Weglaten of `null` telt als" + " 0",
36          minimum = "0",
37          example = "0")
38      Integer gerealiseerd,
39  
40      @Schema(
41          description = "Aantal geplande woningen per jaar van oplevering (fasering), van "
42              + AantallenValidator.EERSTE_JAAR
43              + " tot en met "
44              + AantallenValidator.LAATSTE_JAAR
45              + ". Een jaar weglaten of `null` betekent niet ingevuld, voor het totaal telt dat als 0. Behalve bij"
46              + " sloop moet het totaal gelijk zijn aan `gepland` min `gerealiseerd`",
47          example = "{\"2027\": 10, \"2028\": 10}")
48      SortedMap<Integer, Integer> perJaar) {
49    public Aantal {
50      gepland = Objects.requireNonNullElse(gepland, 0);
51      gerealiseerd = Objects.requireNonNullElse(gerealiseerd, 0);
52      SortedMap<Integer, Integer> filled = new TreeMap<>();
53      if (perJaar != null) {
54        perJaar.entrySet().stream()
55            .filter(e -> e.getKey() != null && e.getValue() != null)
56            .forEach(e -> filled.put(e.getKey(), e.getValue()));
57      }
58      perJaar = filled;
59    }
60  }
