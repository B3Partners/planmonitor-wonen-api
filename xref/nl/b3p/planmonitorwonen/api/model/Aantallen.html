7   package nl.b3p.planmonitorwonen.api.model;
8   
9   import com.fasterxml.jackson.annotation.JsonInclude;
10  import io.swagger.v3.oas.annotations.media.Schema;
11  import java.util.Map;
12  
13  /**
14   * Numbers of dwellings of a planregistratie. The maps are keyed by the category value, for example "Eengezins" for
15   * woningType. A category left out (or null) is not filled in, a category with an {@link Aantal} of 0 is filled in as
16   * zero. For the totals both count as zero. Categories not filled in are left out of the JSON output.
17   *
18   * @param bruto bruto plancapaciteit, the total number of dwellings added by the plan (stored as plancategorie
19   *     "Nieuwbouw")
20   * @param sloop number of dwellings demolished
21   * @param woningType dwellings by woningtype, should add up to bruto
22   * @param wonenEnZorg dwellings by suitability for elderly and care, should add up to bruto
23   * @param flexwoningen dwellings by temporary or permanent, should add up to bruto
24   * @param betaalbaarheid dwellings by price segment, should add up to bruto
25   */
26  @JsonInclude(JsonInclude.Include.NON_EMPTY)
27  @Schema(description = """
28  Aantallen woningen van een planregistratie. Een categorie weglaten of `null` betekent niet ingevuld, een categorie
29  met aantal 0 is ingevuld als nul. Voor de totalen tellen beide als nul. Niet ingevulde categorieën worden in een
30  response weggelaten.
31  
32  Regels:
33  - aantallen zijn niet negatief en `gerealiseerd` is niet meer dan `gepland`
34  - behalve bij sloop is het totaal van `perJaar` gelijk aan `gepland` min `gerealiseerd`
35  - voor `woningType`, `wonenEnZorg`, `flexwoningen` en `betaalbaarheid` is het totaal `gepland` van de categorieën
36  gelijk aan `gepland` van `bruto`, en het totaal per jaar gelijk aan `bruto` in dat jaar. Als `bruto` meer dan 0 is
37  moeten deze groepen dus allemaal ingevuld worden""")
38  public record Aantallen(
39      @Schema(
40          description = "Bruto plancapaciteit: het totaal aantal woningen dat binnen het plan gerealiseerd wordt,"
41              + " door nieuwbouw, transformatie of splitsing")
42      Aantal bruto,
43  
44      @Schema(description = "Aantal woningen dat gesloopt wordt")
45      Aantal sloop,
46  
47      @Schema(description = "Woningen per woningtype") Map<String, Aantal> woningType,
48  
49      @Schema(description = "Woningen per geschiktheid voor ouderen en zorg")
50      Map<String, Aantal> wonenEnZorg,
51  
52      @Schema(description = "Woningen per tijdelijk (flexwoningen) of permanent")
53      Map<String, Aantal> flexwoningen,
54  
55      @Schema(description = "Woningen per prijssegment") Map<String, Aantal> betaalbaarheid) {}
