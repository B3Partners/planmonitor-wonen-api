7   package nl.b3p.planmonitorwonen.api.model;
8   
9   import com.fasterxml.jackson.annotation.JsonInclude;
10  import java.util.Map;
11  
12  /**
13   * Numbers of dwellings of a planregistratie. The maps are keyed by the category value, for example "Eengezins" for
14   * woningType. A category left out (or null) is not filled in, a category with an {@link Aantal} of 0 is filled in as
15   * zero. For the totals both count as zero. Categories not filled in are left out of the JSON output.
16   *
17   * @param bruto bruto plancapaciteit, the total number of dwellings added by the plan (stored as plancategorie
18   *     "Nieuwbouw")
19   * @param sloop number of dwellings demolished
20   * @param woningType dwellings by woningtype, should add up to bruto
21   * @param wonenEnZorg dwellings by suitability for elderly and care, should add up to bruto
22   * @param flexwoningen dwellings by temporary or permanent, should add up to bruto
23   * @param betaalbaarheid dwellings by price segment, should add up to bruto
24   */
25  @JsonInclude(JsonInclude.Include.NON_EMPTY)
26  public record Aantallen(
27      Aantal bruto,
28      Aantal sloop,
29      Map<String, Aantal> woningType,
30      Map<String, Aantal> wonenEnZorg,
31      Map<String, Aantal> flexwoningen,
32      Map<String, Aantal> betaalbaarheid) {}
