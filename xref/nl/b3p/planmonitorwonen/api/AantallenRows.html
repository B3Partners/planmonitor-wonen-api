9   import java.util.ArrayList;
10  import java.util.LinkedHashMap;
11  import java.util.List;
12  import java.util.Map;
13  import java.util.SortedMap;
14  import java.util.TreeMap;
15  import java.util.UUID;
16  import nl.b3p.planmonitorwonen.api.model.Aantal;
17  import nl.b3p.planmonitorwonen.api.model.Aantallen;

19  /**
20   * Converts between {@link Aantallen} and the plancategorie and detailplanning rows in the database. A plancategorie row
21   * has exactly one of the categorie group columns set, the detailplanning rows hold the numbers per year.
22   */
23  final class AantallenRows {
24    private AantallenRows() {}
25  
26    record PlancategorieRow(
27        UUID id,
28        UUID planregistratieId,
29        String nieuwbouw,
30        String woningType,
31        String wonenEnZorg,
32        String flexwoningen,
33        String betaalbaarheid,
34        String sloop,
35        Integer totaalGepland,
36        Integer totaalGerealiseerd) {}
37  
38    record DetailplanningRow(UUID plancategorieId, Integer jaartal, Integer aantalGepland) {}
39  
40    /**
41     * A plancategorie to insert.
42     *
43     * @param column the plancategorie column for the categorie group, a constant safe to use in SQL
44     * @param value the categorie value
45     */
46    record NewPlancategorie(String column, String value, Aantal aantal) {}
47  
48    /**
49     * @param rows the plancategorie rows of a single planregistratie
50     * @param detailplanningen the detailplanning rows by plancategorie id
51     */
52    static Aantallen toAantallen(List<PlancategorieRow> rows, Map<UUID, List<DetailplanningRow>> detailplanningen) {
53      Aantal bruto = null;
54      Aantal sloop = null;
55      Map<String, Aantal> woningType = new LinkedHashMap<>();
56      Map<String, Aantal> wonenEnZorg = new LinkedHashMap<>();
57      Map<String, Aantal> flexwoningen = new LinkedHashMap<>();
58      Map<String, Aantal> betaalbaarheid = new LinkedHashMap<>();
59      for (PlancategorieRow row : rows) {
60        SortedMap<Integer, Integer> perJaar = new TreeMap<>();
61        detailplanningen
62            .getOrDefault(row.id(), List.of())
63            .forEach(d -> perJaar.put(d.jaartal(), d.aantalGepland()));
64        Aantal aantal = new Aantal(row.totaalGepland(), row.totaalGerealiseerd(), perJaar);
65        if (row.nieuwbouw() != null) {
66          bruto = aantal;
67        } else if (row.sloop() != null) {
68          sloop = aantal;
69        } else if (row.woningType() != null) {
70          woningType.put(row.woningType(), aantal);
71        } else if (row.wonenEnZorg() != null) {
72          wonenEnZorg.put(row.wonenEnZorg(), aantal);
73        } else if (row.flexwoningen() != null) {
74          flexwoningen.put(row.flexwoningen(), aantal);
75        } else if (row.betaalbaarheid() != null) {
76          betaalbaarheid.put(row.betaalbaarheid(), aantal);
77        }
78      }
79      return new Aantallen(bruto, sloop, woningType, wonenEnZorg, flexwoningen, betaalbaarheid);
80    }
81  
82    static List<NewPlancategorie> toPlancategorieen(Aantallen aantallen) {
83      List<NewPlancategorie> result = new ArrayList<>();
84      if (aantallen == null) {
85        return result;
86      }
87      if (aantallen.bruto() != null) {
88        result.add(new NewPlancategorie("nieuwbouw", "Nieuwbouw", aantallen.bruto()));
89      }
90      if (aantallen.sloop() != null) {
91        result.add(new NewPlancategorie("sloop", "Sloop", aantallen.sloop()));
92      }
93      addGroep(result, "woning_type", aantallen.woningType());
94      addGroep(result, "wonen_en_zorg", aantallen.wonenEnZorg());
95      addGroep(result, "flexwoningen", aantallen.flexwoningen());
96      addGroep(result, "betaalbaarheid", aantallen.betaalbaarheid());
97      return result;
98    }
99  
100   private static void addGroep(List<NewPlancategorie> result, String column, Map<String, Aantal> groep) {
101     if (groep != null) {
102       groep.forEach((value, aantal) -> {
103         if (aantal != null) {
104           result.add(new NewPlancategorie(column, value, aantal));
105         }
106       });
107     }
108   }
