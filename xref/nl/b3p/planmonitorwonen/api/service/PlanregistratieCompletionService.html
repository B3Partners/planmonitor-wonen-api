7   package nl.b3p.planmonitorwonen.api.service;

9   import java.lang.invoke.MethodHandles;
10  import java.util.List;
11  import nl.b3p.planmonitorwonen.api.PlanmonitorWonenDatabaseService;
12  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
13  import nl.b3p.planmonitorwonen.api.validation.PlanregistratieValidationException;
14  import nl.b3p.planmonitorwonen.api.validation.ValidationError;
15  import org.locationtech.jts.geom.Geometry;
16  import org.locationtech.jts.io.ParseException;
17  import org.locationtech.jts.io.WKTReader;
18  import org.slf4j.Logger;
19  import org.slf4j.LoggerFactory;
20  import org.springframework.context.annotation.Profile;
21  import org.springframework.stereotype.Service;

23  /**
24   * Fills in fields of a planregistratie the server can determine: the provincie of the gemeente, a default
25   * vertrouwelijkheid and the regio and woonmilieu from the geometry when not set by the client. Also checks the gemeente
26   * exists and the geometry lies (partly) within the gemeente.
27   */
28  @Service
29  @Profile("!test")
30  public class PlanregistratieCompletionService {
31    private static final Logger logger =
32        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
33  
34    private final PlanmonitorWonenDatabaseService pmwDb;
35    private final PlanregistratieAutofillService autofillService;
36  
37    public PlanregistratieCompletionService(
38        PlanmonitorWonenDatabaseService pmwDb, PlanregistratieAutofillService autofillService) {
39      this.pmwDb = pmwDb;
40      this.autofillService = autofillService;
41    }
42  
43    /**
44     * Fills in the fields the server can determine.
45     *
46     * @throws PlanregistratieValidationException for an unknown gemeente or a geometry outside the gemeente
47     * @param planregistratie a planregistratie which passed
48     *     {@link nl.b3p.planmonitorwonen.api.validation.PlanregistratieValidator}
49     */
50    public void complete(Planregistratie planregistratie) throws ParseException {
51      String provincie = pmwDb.getProvincieForGemeente(planregistratie.getGemeente())
52          .orElseThrow(() -> new PlanregistratieValidationException(List.of(
53              new ValidationError("gemeente", "Onbekende gemeente: " + planregistratie.getGemeente()))));
54      planregistratie.setProvincie(provincie);
55  
56      if (!pmwDb.geometryIntersectsGemeente(planregistratie.getGemeente(), planregistratie.getGeometrie())
57          .orElse(false)) {
58        throw new PlanregistratieValidationException(List.of(new ValidationError(
59            "geometrie", "Geometrie ligt niet in gemeente " + planregistratie.getGemeente())));
60      }
61  
62      if (isBlank(planregistratie.getVertrouwelijkheid())) {
63        planregistratie.setVertrouwelijkheid("Openbaar");
64      }
65  
66      if (isBlank(planregistratie.getRegio()) || isBlank(planregistratie.getBeoogdWoonmilieuAbf13())) {
67        Geometry geometry = new WKTReader().read(planregistratie.getGeometrie());
68        if (isBlank(planregistratie.getRegio())) {
69          planregistratie.setRegio(first(autofillService.getRegios(geometry), "regio", planregistratie));
70        }
71        if (isBlank(planregistratie.getBeoogdWoonmilieuAbf13())) {
72          planregistratie.setBeoogdWoonmilieuAbf13(
73              first(autofillService.getWoonmilieus(geometry), "woonmilieu", planregistratie));
74        }
75      }
76    }
77  
78    private static String first(List<String> values, String what, Planregistratie planregistratie) {
79      if (values.isEmpty()) {
80        logger.warn(
81            "No {} found for geometry of plan id {}, name \"{}\", not set (WFS features not loaded?)",
82            what,
83            planregistratie.getId(),
84            planregistratie.getPlanNaam());
85        return null;
86      }
87      return values.getFirst();
88    }
89  
90    private static boolean isBlank(String s) {
91      return s == null || s.isBlank();
92    }
93  }
