7   package nl.b3p.planmonitorwonen.api.service;

9   import java.lang.invoke.MethodHandles;
10  import java.util.List;
11  import nl.b3p.planmonitorwonen.api.PlanmonitorWonenDatabaseService;
12  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
13  import nl.b3p.planmonitorwonen.api.validation.PlanregistratieValidationException;
14  import nl.b3p.planmonitorwonen.api.validation.ValidationError;
15  import org.locationtech.jts.geom.Geometry;
16  import org.locationtech.jts.io.ParseException;
17  import org.locationtech.jts.io.WKTReader;
18  import org.slf4j.Logger;
19  import org.slf4j.LoggerFactory;
20  import org.springframework.context.annotation.Profile;
21  import org.springframework.stereotype.Service;
22  
23  /**
24   * Fills in fields of a planregistratie the server can determine: the provincie of the gemeente, a default
25   * vertrouwelijkheid and the regio and woonmilieu from the geometry when not set by the client.
26   */
27  @Service
28  @Profile("!test")
29  public class PlanregistratieCompletionService {
30    private static final Logger logger =
31        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
32  
33    private final PlanmonitorWonenDatabaseService pmwDb;
34    private final PlanregistratieAutofillService autofillService;
35  
36    public PlanregistratieCompletionService(
37        PlanmonitorWonenDatabaseService pmwDb, PlanregistratieAutofillService autofillService) {
38      this.pmwDb = pmwDb;
39      this.autofillService = autofillService;
40    }
41  
42    /**
43     * Fills in the fields the server can determine.
44     *
45     * @param planregistratie a planregistratie which passed
46     *     {@link nl.b3p.planmonitorwonen.api.validation.PlanregistratieValidator}
47     */
48    public void complete(Planregistratie planregistratie) throws ParseException {
49      String provincie = pmwDb.getProvincieForGemeente(planregistratie.getGemeente())
50          .orElseThrow(() -> new PlanregistratieValidationException(List.of(
51              new ValidationError("gemeente", "Onbekende gemeente: " + planregistratie.getGemeente()))));
52      planregistratie.setProvincie(provincie);
53  
54      if (isBlank(planregistratie.getVertrouwelijkheid())) {
55        planregistratie.setVertrouwelijkheid("Openbaar");
56      }
57  
58      if (isBlank(planregistratie.getRegio()) || isBlank(planregistratie.getBeoogdWoonmilieuAbf13())) {
59        Geometry geometry = new WKTReader().read(planregistratie.getGeometrie());
60        if (isBlank(planregistratie.getRegio())) {
61          planregistratie.setRegio(first(autofillService.getRegios(geometry), "regio", planregistratie));
62        }
63        if (isBlank(planregistratie.getBeoogdWoonmilieuAbf13())) {
64          planregistratie.setBeoogdWoonmilieuAbf13(
65              first(autofillService.getWoonmilieus(geometry), "woonmilieu", planregistratie));
66        }
67      }
68    }
69  
70    private static String first(List<String> values, String what, Planregistratie planregistratie) {
71      if (values.isEmpty()) {
72        logger.warn(
73            "No {} found for geometry of plan id {}, name \"{}\", not set (WFS features not loaded?)",
74            what,
75            planregistratie.getId(),
76            planregistratie.getPlanNaam());
77        return null;
78      }
79      return values.getFirst();
80    }
81  
82    private static boolean isBlank(String s) {
83      return s == null || s.isBlank();
84    }
85  }
