7   package nl.b3p.planmonitorwonen.api.validation;
8   
9   import java.util.List;
10  import java.util.Map;
11  
12  /**
13   * The values of the enum types in the database, so the validators can report an unknown value with the possible values
14   * instead of the database rejecting it. Read from the database at startup by
15   * {@link nl.b3p.planmonitorwonen.api.PlanmonitorWonenDatabaseService#getEnumValues()}, so the database schema is the
16   * only place where the values are defined.
17   *
18   * @param values the values in enum sort order by enum type name
19   */
20  public record EnumValues(Map<String, List<String>> values) {
21    public static final String PLANTYPE = "pmw_plantype";
22    public static final String STATUS_PLANOLOGISCH = "pmw_planologisch_status";
23    public static final String VERTROUWELIJKHEID = "pmw_vertrouwelijkheid";
24    public static final String OPDRACHTGEVER_TYPE = "pmw_opdrachtgever";
25    public static final String STATUS_PROJECT = "pmw_project_status";
26    public static final String KNELPUNTEN_MEERKEUZE = "pmw_knelpunten_meerkeuze";
27    public static final String BEOOGD_WOONMILIEU_ABF13 = "pmw_woonmilieu_abf13";
28    public static final String WONING_TYPE = "pmw_woning_type";
29    public static final String WONEN_EN_ZORG = "pmw_wonen_en_zorg";
30    public static final String FLEXWONINGEN = "pmw_flexwoningen";
31    public static final String BETAALBAARHEID = "pmw_betaalbaarheid";
32  
33    public EnumValues {
34      values = Map.copyOf(values);
35    }
36  
37    /**
38     * Returns the values of an enum type in enum sort order.
39     *
40     * @throws IllegalStateException when the enum type does not exist in the database
41     */
42    public List<String> get(String enumType) {
43      List<String> result = values.get(enumType);
44      if (result == null) {
45        throw new IllegalStateException("Enum type %s not found in database".formatted(enumType));
46      }
47      return result;
48    }
49  }
