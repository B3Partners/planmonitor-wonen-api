7   package nl.b3p.planmonitorwonen.api.validation;
8   
9   import java.util.LinkedHashMap;
10  import java.util.List;
11  import java.util.Map;
12  import java.util.Objects;
13  import java.util.SortedSet;
14  import java.util.TreeSet;
15  import nl.b3p.planmonitorwonen.api.model.Aantal;
16  import nl.b3p.planmonitorwonen.api.model.Aantallen;
17  
18  /**
19   * Validates the aantallen of a planregistratie with the same rules as the categorie table in the frontend. A categorie
20   * or year not filled in counts as zero.
21   *
22   * <ul>
23   *   <li>the numbers are not negative and gerealiseerd is not more than gepland
24   *   <li>the years are within the years shown in the frontend
25   *   <li>for each categorie except sloop the numbers per year add up to gepland minus gerealiseerd
26   *   <li>for each categorie group the gepland numbers add up to bruto
27   *   <li>for each categorie group and year the numbers add up to the number of bruto for that year
28   * </ul>
29   */
30  public final class AantallenValidator {
31    /** The years shown in the frontend, for the first 10 years per year and then two periods of 5 years. */
32    public static final int EERSTE_JAAR = 2024;
33  
34    public static final int LAATSTE_JAAR = 2043;
35  
36    /** The enum type in the database with the categorie values per group. */
37    static final Map<String, String> GROEP_ENUM_TYPES = Map.of(
38        "woningType", EnumValues.WONING_TYPE,
39        "wonenEnZorg", EnumValues.WONEN_EN_ZORG,
40        "flexwoningen", EnumValues.FLEXWONINGEN,
41        "betaalbaarheid", EnumValues.BETAALBAARHEID);
42  
43    private static final String BRUTO_FIELD = "aantallen.bruto";
44  
45    private AantallenValidator() {}
46  
47    public static void validate(Aantallen aantallen, EnumValues enumValues, List<ValidationError> errors) {
48      if (aantallen == null) {
49        return;
50      }
51      Aantal bruto = aantallen.bruto();
52      boolean valuesValid = validateAantal(BRUTO_FIELD, bruto, true, errors);
53      valuesValid &= validateAantal("aantallen.sloop", aantallen.sloop(), false, errors);
54  
55      Map<String, Map<String, Aantal>> groepen = new LinkedHashMap<>();
56      groepen.put("woningType", aantallen.woningType());
57      groepen.put("wonenEnZorg", aantallen.wonenEnZorg());
58      groepen.put("flexwoningen", aantallen.flexwoningen());
59      groepen.put("betaalbaarheid", aantallen.betaalbaarheid());
60  
61      for (Map.Entry<String, Map<String, Aantal>> groep : groepen.entrySet()) {
62        Map<String, Aantal> categorieen = Objects.requireNonNullElse(groep.getValue(), Map.of());
63        List<String> categorieValues = enumValues.get(GROEP_ENUM_TYPES.get(groep.getKey()));
64        for (Map.Entry<String, Aantal> categorie : categorieen.entrySet()) {
65          String field = "aantallen." + groep.getKey() + "." + categorie.getKey();
66          if (!categorieValues.contains(categorie.getKey())) {
67            errors.add(new ValidationError(
68                field,
69                "Onbekende categorie \"%s\", mogelijke waarden: %s"
70                    .formatted(categorie.getKey(), String.join(", ", categorieValues))));
71            valuesValid = false;
72          }
73          valuesValid &= validateAantal(field, categorie.getValue(), true, errors);
74        }
75      }
76      if (!valuesValid) {
77        // The totals are only compared when the values themselves are valid
78        return;
79      }
80  
81      int brutoGepland = bruto == null ? 0 : bruto.gepland();
82      for (Map.Entry<String, Map<String, Aantal>> groep : groepen.entrySet()) {
83        Map<String, Aantal> categorieen = Objects.requireNonNullElse(groep.getValue(), Map.of());
84        String field = "aantallen." + groep.getKey();
85        int totaal = categorieen.values().stream()
86            .filter(Objects::nonNull)
87            .mapToInt(Aantal::gepland)
88            .sum();
89        if (totaal != brutoGepland) {
90          errors.add(new ValidationError(
91              field,
92              "Totaal gepland van %s (%d) is niet gelijk aan bruto (%d)"
93                  .formatted(groep.getKey(), totaal, brutoGepland)));
94        }
95  
96        SortedSet<Integer> jaren = new TreeSet<>();
97        if (bruto != null) {
98          jaren.addAll(bruto.perJaar().keySet());
99        }
100       categorieen.values().stream()
101           .filter(Objects::nonNull)
102           .forEach(a -> jaren.addAll(a.perJaar().keySet()));
103       for (Integer jaar : jaren) {
104         int brutoJaar = bruto == null ? 0 : bruto.perJaar().getOrDefault(jaar, 0);
105         int jaarTotaal = categorieen.values().stream()
106             .filter(Objects::nonNull)
107             .mapToInt(a -> a.perJaar().getOrDefault(jaar, 0))
108             .sum();
109         if (jaarTotaal != brutoJaar) {
110           errors.add(new ValidationError(
111               field + ".perJaar." + jaar,
112               "Totaal van %s in %d (%d) is niet gelijk aan bruto in %d (%d)"
113                   .formatted(groep.getKey(), jaar, jaarTotaal, jaar, brutoJaar)));
114         }
115       }
116     }
117   }
118 
119   /**
120    * Validates the values of an aantal and whether the numbers per year add up.
121    *
122    * @param restcapaciteitCheck whether the numbers per year must add up to gepland minus gerealiseerd
123    * @return whether the values are valid, so the totals can be compared, regardless of the numbers per year adding up
124    */
125   private static boolean validateAantal(
126       String field, Aantal aantal, boolean restcapaciteitCheck, List<ValidationError> errors) {
127     if (aantal == null) {
128       return true;
129     }
130     int errorCount = errors.size();
131     if (aantal.gepland() < 0) {
132       errors.add(new ValidationError(field + ".gepland", "Gepland mag niet negatief zijn"));
133     }
134     if (aantal.gerealiseerd() < 0) {
135       errors.add(new ValidationError(field + ".gerealiseerd", "Gerealiseerd mag niet negatief zijn"));
136     }
137     if (aantal.gepland() >= 0 && aantal.gerealiseerd() > aantal.gepland()) {
138       errors.add(new ValidationError(
139           field + ".gerealiseerd",
140           "Gerealiseerd (%d) mag niet meer zijn dan gepland (%d)"
141               .formatted(aantal.gerealiseerd(), aantal.gepland())));
142     }
143     aantal.perJaar().forEach((jaar, aantalGepland) -> {
144       if (jaar < EERSTE_JAAR || jaar > LAATSTE_JAAR) {
145         errors.add(new ValidationError(
146             field + ".perJaar." + jaar,
147             "Jaar %d moet tussen %d en %d liggen".formatted(jaar, EERSTE_JAAR, LAATSTE_JAAR)));
148       }
149       if (aantalGepland < 0) {
150         errors.add(new ValidationError(field + ".perJaar." + jaar, "Aantal mag niet negatief zijn"));
151       }
152     });
153     if (errors.size() > errorCount) {
154       return false;
155     }
156     if (restcapaciteitCheck) {
157       int restcapaciteit = aantal.gepland() - aantal.gerealiseerd();
158       int perJaarTotaal = aantal.perJaar().values().stream()
159           .mapToInt(Integer::intValue)
160           .sum();
161       if (perJaarTotaal != restcapaciteit) {
162         errors.add(new ValidationError(
163             field + ".perJaar",
164             "Totaal per jaar (%d) is niet gelijk aan gepland min gerealiseerd (%d)"
165                 .formatted(perJaarTotaal, restcapaciteit)));
166       }
167     }
168     return true;
169   }
