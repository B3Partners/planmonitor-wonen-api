7   package nl.b3p.planmonitorwonen.api.validation;
8   
9   import java.util.ArrayList;
10  import java.util.List;
11  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
12  import org.locationtech.jts.geom.Geometry;
13  import org.locationtech.jts.geom.Polygon;
14  import org.locationtech.jts.io.ParseException;
15  import org.locationtech.jts.io.WKTReader;
16  
17  /**
18   * Validates a planregistratie received from a client. Only the fields of the basisset 2.0 and the fields needed to
19   * identify and authorize the planregistratie are required, the provincie specific fields are optional because
20   * planregistraties may be delivered by software of a gemeente.
21   */
22  public final class PlanregistratieValidator {
23    private PlanregistratieValidator() {}
24  
25    public static List<ValidationError> validate(Planregistratie planregistratie) {
26      List<ValidationError> errors = new ArrayList<>();
27      required(errors, "planNaam", "Plannaam", planregistratie.getPlanNaam());
28      required(errors, "gemeente", "Gemeente", planregistratie.getGemeente());
29      required(errors, "plantype", "Plantype", planregistratie.getPlantype());
30      required(errors, "statusPlanologisch", "Planologische planstatus", planregistratie.getStatusPlanologisch());
31      validateGeometrie(errors, planregistratie.getGeometrie());
32      AantallenValidator.validate(planregistratie.getAantallen(), errors);
33      return errors;
34    }
35  
36    private static void required(List<ValidationError> errors, String field, String label, String value) {
37      if (value == null || value.isBlank()) {
38        errors.add(new ValidationError(field, label + " is verplicht"));
39      }
40    }
41  
42    private static void validateGeometrie(List<ValidationError> errors, String wkt) {
43      if (wkt == null || wkt.isBlank()) {
44        errors.add(new ValidationError("geometrie", "Geometrie is verplicht"));
45        return;
46      }
47      Geometry geometry;
48      try {
49        geometry = new WKTReader().read(wkt);
50      } catch (ParseException e) {
51        errors.add(new ValidationError("geometrie", "Geometrie is geen geldige WKT: " + e.getMessage()));
52        return;
53      }
54      if (!(geometry instanceof Polygon)) {
55        errors.add(new ValidationError(
56            "geometrie", "Geometrie moet een polygoon zijn, geen " + geometry.getGeometryType()));
57      }
58    }
59  }
