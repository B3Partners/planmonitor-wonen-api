7   package nl.b3p.planmonitorwonen.api.configuration;
8   
9   import org.springframework.beans.factory.annotation.Value;
10  import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
11  import org.springframework.context.annotation.Configuration;
12  import org.springframework.web.servlet.config.annotation.ViewControllerRegistry;
13  import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;
14  
15  /**
16   * Redirects the base path of the API to the Swagger UI, so opening it in a browser shows the API documentation. Not
17   * done with an error view for paths that are not found, because the ProblemDetail exception handler answers those
18   * before the error page is reached.
19   */
20  @Configuration
21  @ConditionalOnProperty(name = "springdoc.swagger-ui.enabled", matchIfMissing = true)
22  public class SwaggerUiRedirectConfiguration implements WebMvcConfigurer {
23    private final String basePath;
24  
25    private final String swaggerUiPath;
26  
27    public SwaggerUiRedirectConfiguration(
28        @Value("${planmonitor-wonen-api.base-path}") String basePath,
29        @Value("${springdoc.swagger-ui.path:/swagger-ui.html}") String swaggerUiPath) {
30      this.basePath = basePath;
31      this.swaggerUiPath = swaggerUiPath;
32    }
33  
34    @Override
35    public void addViewControllers(ViewControllerRegistry registry) {
36      registry.addRedirectViewController(basePath, swaggerUiPath);
37      registry.addRedirectViewController(basePath + "/", swaggerUiPath);
38    }
39  }
