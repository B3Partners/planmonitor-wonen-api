72    /** The Planregistratie properties with the values of an enum type in the database. */
73    private static final Map<String, String> PLANREGISTRATIE_ENUM_TYPES = Map.of(
74        "plantype", EnumValues.PLANTYPE,
75        "statusPlanologisch", EnumValues.STATUS_PLANOLOGISCH,
76        "vertrouwelijkheid", EnumValues.VERTROUWELIJKHEID,

81    @Bean
82    public OpenAPI openAPI() throws IOException {
83      @SuppressWarnings("rawtypes")
84      Map<String, Schema> problemDetailSchemas =
85          ModelConverters.getInstance(true).readAll(ProblemDetailSchema.class);
86      Content problemDetail =
87          new Content().addMediaType(PROBLEM_JSON, new MediaType().schema(new Schema<>().$ref("ProblemDetail")));
88  
89      return new OpenAPI()
90          .info(new Info().title("Planmonitor Wonen API").version(VERSION).description(description()))
91          .addSecurityItem(new SecurityRequirement().addList(SESSION_SECURITY_SCHEME))
92          .components(new Components()
93              .addSecuritySchemes(
94                  SESSION_SECURITY_SCHEME,
95                  new SecurityScheme()
96                      .type(SecurityScheme.Type.APIKEY)
97                      .in(SecurityScheme.In.COOKIE)
98                      .name("SESSION")
99                      .description("Sessie-cookie van het inloggen bij Tailormap, zie Inloggen"))
100             .schemas(problemDetailSchemas)
101             .addResponses(
102                 UNAUTHORIZED,
103                 new ApiResponse()
104                     .description("Niet ingelogd of de sessie is verlopen")
105                     .content(problemDetail))
106             .addResponses(
107                 FORBIDDEN,
108                 new ApiResponse()
109                     .description(
110                         "Geen toegang: de gebruiker heeft geen autorisatie voor de gemeente van de"
111                             + " planregistratie of voor Planmonitor Wonen")
112                     .content(problemDetail))
113             .addResponses(
114                 NOT_FOUND,
115                 new ApiResponse()
116                     .description(
117                         "Planregistratie niet gevonden, of van een gemeente waarvoor de gebruiker geen"
118                             + " autorisatie heeft")
119                     .content(problemDetail))
120             .addExamples(
121                 "Planregistratie",
122                 new Example()
123                     .summary("Planregistratie met aantallen")
124                     .value(readJson(PLANREGISTRATIE_EXAMPLE_RESOURCE))));
125   }

127   /**
128    * Sets the enum values read from the database on the model properties. This can't be done with
129    * {@code @Schema(allowableValues)} without duplicating the values of the database schema.
130    */
131   @Bean
132   @Profile("!test")
133   public OpenApiCustomizer enumValuesCustomizer(EnumValues enumValues) {
134     return openApi -> setEnumValues(openApi, enumValues);
135   }
136 
137   /** The swagger models use raw types for the properties of a schema. */
138   @SuppressWarnings({"rawtypes", "unchecked"})
139   private static void setEnumValues(OpenAPI openApi, EnumValues enumValues) {
140     Map<String, Schema> schemas = openApi.getComponents().getSchemas();
141 
142     Map<String, Schema> planregistratie = schemas.get("Planregistratie").getProperties();
143     PLANREGISTRATIE_ENUM_TYPES.forEach(
144         (property, enumType) -> setEnum(planregistratie.get(property), enumValues.get(enumType)));
145     setEnum(planregistratie.get("knelpuntenMeerkeuze").getItems(), enumValues.get(EnumValues.KNELPUNTEN_MEERKEUZE));
146 
147     Map<String, Schema> aantallen = schemas.get("Aantallen").getProperties();
148     AantallenValidator.GROEP_ENUM_TYPES.forEach(
149         (groep, enumType) -> setCategorieen(aantallen.get(groep), enumValues.get(enumType)));
150   }
151 
152   @SuppressWarnings("unchecked")
153   private static void setEnum(Schema<?> schema, List<String> values) {
154     ((Schema<String>) schema).setEnum(new ArrayList<>(values));
155   }
156 
157   /**
158    * Replaces the additionalProperties of a categorie group map with a property for each categorie, so the possible
159    * keys are visible in the documentation.
160    */
161   @SuppressWarnings("rawtypes")
162   private static void setCategorieen(Schema<?> groep, List<String> categorieen) {
163     Map<String, Schema> properties = new LinkedHashMap<>();
164     for (String categorie : categorieen) {
165       properties.put(categorie, new Schema<>().$ref("Aantal"));
166     }
167     groep.setProperties(properties);
168     groep.setAdditionalProperties(false);
169   }

171   private static String description() throws IOException {
172     return new ClassPathResource(DESCRIPTION_RESOURCE)
173         .getContentAsString(StandardCharsets.UTF_8)
174         .replace("{majorVersion}", MAJOR_VERSION)
175         .replace("{version}", VERSION);
176   }
177 
178   private static Object readJson(String resource) throws IOException {
179     return Json.mapper()
180         .readValue(new ClassPathResource(resource).getContentAsString(StandardCharsets.UTF_8), Object.class);
181   }
