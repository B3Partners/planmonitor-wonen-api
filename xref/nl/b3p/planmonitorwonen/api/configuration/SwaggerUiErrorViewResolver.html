7   package nl.b3p.planmonitorwonen.api.configuration;
8   
9   import jakarta.servlet.http.HttpServletRequest;
10  import java.util.Map;
11  import org.jspecify.annotations.NonNull;
12  import org.springdoc.core.properties.SwaggerUiConfigProperties;
13  import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
14  import org.springframework.boot.webmvc.autoconfigure.error.ErrorViewResolver;
15  import org.springframework.http.HttpStatus;
16  import org.springframework.stereotype.Component;
17  import org.springframework.web.servlet.ModelAndView;
18  
19  /**
20   * Redirects browsers to the Swagger UI instead of showing the Whitelabel Error Page for paths that are not found. Only
21   * used for requests accepting HTML, API clients still get a JSON error response.
22   */
23  @Component
24  @ConditionalOnProperty(name = "springdoc.swagger-ui.enabled", matchIfMissing = true)
25  public class SwaggerUiErrorViewResolver implements ErrorViewResolver {
26    private final SwaggerUiConfigProperties swaggerUiConfigProperties;
27  
28    public SwaggerUiErrorViewResolver(SwaggerUiConfigProperties swaggerUiConfigProperties) {
29      this.swaggerUiConfigProperties = swaggerUiConfigProperties;
30    }
31  
32    @Override
33    public ModelAndView resolveErrorView(
34        @NonNull HttpServletRequest request, @NonNull HttpStatus status, @NonNull Map<String, Object> model) {
35      if (status == HttpStatus.NOT_FOUND) {
36        return new ModelAndView("redirect:" + swaggerUiConfigProperties.getPath());
37      }
38      return null;
39    }
40  }
