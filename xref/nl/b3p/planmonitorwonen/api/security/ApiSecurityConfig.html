6   package nl.b3p.planmonitorwonen.api.security;
7   
8   import jakarta.servlet.http.HttpServletResponse;
9   import java.lang.invoke.MethodHandles;
10  import org.slf4j.Logger;
11  import org.slf4j.LoggerFactory;
12  import org.springframework.beans.factory.annotation.Value;
13  import org.springframework.context.annotation.Bean;
14  import org.springframework.context.annotation.Configuration;
15  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
16  import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
17  import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
18  import org.springframework.security.config.http.SessionCreationPolicy;
19  import org.springframework.security.web.SecurityFilterChain;
20  import org.springframework.security.web.savedrequest.NullRequestCache;
21  import org.springframework.security.web.savedrequest.RequestCache;
22  
23  @Configuration
24  @EnableWebSecurity(debug = false)
25  public class ApiSecurityConfig {
26    private static final Logger logger =
27        LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
28  
29    @Value("${management.endpoints.web.base-path}")
30    private String actuatorBasePath;
31  
32    @Bean
33    public SecurityFilterChain apiFilterChain(HttpSecurity http) throws Exception {
34      RequestCache nullRequestCache = new NullRequestCache();
35      http.csrf(AbstractHttpConfigurer::disable)
36          .authorizeHttpRequests(authorize -> {
37            authorize.requestMatchers(actuatorBasePath + "/health/**").permitAll();
38            authorize
39                .requestMatchers(actuatorBasePath, actuatorBasePath + "/**")
40                .hasAnyAuthority("admin", "actuator");
41            // API endpoints check authorization themselves with PlanmonitorAuthenticationService
42            authorize.anyRequest().permitAll();
43          })
44          .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.NEVER))
45          .requestCache((cache) -> cache.requestCache(nullRequestCache))
46          .exceptionHandling(httpSecurityExceptionHandlingConfigurer ->
47              httpSecurityExceptionHandlingConfigurer.authenticationEntryPoint(
48                  (request, response, authException) -> {
49                    logger.debug("Unauthorized request: {}", request, authException);
50                    response.sendError(
51                        HttpServletResponse.SC_UNAUTHORIZED, authException.getLocalizedMessage());
52                  }));
53  
54      return http.build();
55    }
56  }
