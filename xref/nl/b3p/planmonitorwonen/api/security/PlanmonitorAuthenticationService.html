7   package nl.b3p.planmonitorwonen.api.security;
8   
9   import static org.springframework.http.HttpStatus.FORBIDDEN;
10  import static org.springframework.http.HttpStatus.UNAUTHORIZED;
11  
12  import java.util.Set;
13  import java.util.stream.Collectors;
14  import org.springframework.context.annotation.Profile;
15  import org.springframework.security.authentication.AnonymousAuthenticationToken;
16  import org.springframework.security.core.Authentication;
17  import org.springframework.security.core.context.SecurityContextHolder;
18  import org.springframework.stereotype.Service;
19  import org.springframework.web.server.ResponseStatusException;
20  import org.tailormap.api.security.TailormapUserDetails;
21  
22  @Service
23  @Profile("!test")
24  public class PlanmonitorAuthenticationService {
25    public record PlanmonitorAuthentication(
26        Authentication authentication, boolean isProvincie, Set<String> gemeentes) {}
27  
28    /**
29     * Requires a logged-in Tailormap user, without requiring authorization for a gemeente or provincie (for data a
30     * Tailormap admin also needs, such as the gemeente choices for the group properties).
31     */
32    public Authentication requireAuthenticated() throws ResponseStatusException {
33      final Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
34      // Without a session the authentication is an anonymous authentication instead of null
35      if (authentication == null || authentication instanceof AnonymousAuthenticationToken) {
36        throw new ResponseStatusException(UNAUTHORIZED, "Niet ingelogd");
37      }
38      return authentication;
39    }
40  
41    /**
42     * Requires a logged-in Tailormap user with a group property typeGebruiker "provincie" or one or more gemeente group
43     * properties.
44     *
45     * @return the provincie flag and the gemeentes the user is authorized for
46     * @throws ResponseStatusException 401 when not logged in, 403 when not authorized for a gemeente or provincie
47     */
48    public PlanmonitorAuthentication requirePlanmonitorUser() throws ResponseStatusException {
49      final Authentication authentication = requireAuthenticated();
50  
51      if (authentication.getPrincipal() instanceof TailormapUserDetails userDetails) {
52        // these keys have been registered in the TM API, see:
53        // https://github.com/Tailormap/tailormap-api/blob/main/src/main/java/org/tailormap/api/persistence/helper/AdminAdditionalPropertyHelper.java#L18-L24
54  
55        boolean isProvincie =
56            userDetails.streamAllPropertiesForKey("typeGebruiker").anyMatch("provincie"::equals);
57  
58        Set<String> gemeentes = userDetails
59            .streamAllPropertiesForKey("gemeente")
60            .map(Object::toString)
61            .collect(Collectors.toSet());
62        PlanmonitorAuthentication result = new PlanmonitorAuthentication(authentication, isProvincie, gemeentes);
63  
64        if (!result.isProvincie && result.gemeentes.isEmpty()) {
65          throw new ResponseStatusException(
66              FORBIDDEN, "Gebruiker is niet geautoriseerd voor een gemeente of provincie");
67        }
68        return result;
69      } else {
70        throw new ResponseStatusException(FORBIDDEN, "Gebruiker is niet geautoriseerd voor Planmonitor Wonen");
71      }
72    }
73  }
