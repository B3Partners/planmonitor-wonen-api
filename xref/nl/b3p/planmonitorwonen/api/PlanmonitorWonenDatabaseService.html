55    public PlanmonitorWonenDatabaseService(JdbcClient jdbcClient) {
56      this.jdbcClient = jdbcClient;
57  
58      final GenericConversionService conversionService = new GenericConversionService();
59      DefaultConversionService.addDefaultConverters(conversionService);
60      conversionService.addConverter(new Converter<PgArray, String[]>() {
61        @Override
62        public String[] convert(PgArray source) {
63          try {
64            return (String[]) source.getArray();
65          } catch (SQLException e) {
66            throw new RuntimeException(e);
67          }
68        }
69      });
70  
71      planregistratieRowMapper = new SimplePropertyRowMapper<>(Planregistratie.class, conversionService) {
72        @Override
73        @NonNull public Planregistratie mapRow(@NonNull ResultSet rs, int rowNumber) throws SQLException {
74          Planregistratie planregistratie = super.mapRow(rs, rowNumber);
75          planregistratie.setGeometrie(wkbToWkt(planregistratie.getGeometrie()));
76          return planregistratie;
77        }
78      };
79    }

81    private String wkbToWkt(String wkb) {
82      if (wkb == null) {
83        return null;
84      }
85      try {
86        Geometry geometry = new WKBReader().read(WKBReader.hexToBytes(wkb));
87        return new WKTWriter().write(geometry);
88      } catch (ParseException e) {
89        throw new RuntimeException(e);
90      }
91    }
92  
93    private byte[] wktToWkb(String wkt) throws ParseException {
94      Geometry geometry = new WKTReader().read(wkt);
95      return new WKBWriter().write(geometry);
96    }

119   /**
120    * @param planregistratieFilter SQL condition for the planregistratie table, only use with constant strings and use
121    *     parameters for values
122    */
123   private List<Planregistratie> getPlanregistraties(
124       String planregistratieFilter, Map<String, ?> params, boolean withAantallen) {
125     List<Planregistratie> planregistraties = jdbcClient
126         .sql("select * from planregistratie where %s order by plan_naam".formatted(planregistratieFilter))
127         .params(params)
128         .query(planregistratieRowMapper)
129         .list();
130     if (!withAantallen || planregistraties.isEmpty()) {
131       return planregistraties;
132     }
133 
134     String planregistratieIds = "select id from planregistratie where " + planregistratieFilter;
135     Map<UUID, List<DetailplanningRow>> detailplanningen =
136         jdbcClient.sql("""
137 select d.plancategorie_id, d.jaartal, d.aantal_gepland
138 from detailplanning d
139 join plancategorie c on c.id = d.plancategorie_id
140 where c.planregistratie_id in (%s)""".formatted(planregistratieIds)).params(params).query(DetailplanningRow.class).stream()
141             .collect(groupingBy(DetailplanningRow::plancategorieId));
142 
143     // Ordered by the categorie values in enum order
144     Map<UUID, List<PlancategorieRow>> plancategorieen =
145         jdbcClient.sql("""
146 select * from plancategorie
147 where planregistratie_id in (%s)
148 order by nieuwbouw, woning_type, wonen_en_zorg, flexwoningen, betaalbaarheid, sloop""".formatted(planregistratieIds)).params(params).query(PlancategorieRow.class).stream()
149             .collect(groupingBy(PlancategorieRow::planregistratieId));
150 
151     planregistraties.forEach(p -> p.setAantallen(AantallenRows.toAantallen(
152         plancategorieen.getOrDefault(UUID.fromString(p.getId()), List.of()), detailplanningen)));
153     return planregistraties;
154   }

166   /**
167    * Inserts a planregistratie or replaces an existing one, including the aantallen. The creator, createdAt, editor
168    * and editedAt fields of the planregistratie are set from the username and the current time, the values from the
169    * client are ignored. For an existing planregistratie the creator and createdAt are kept.
170    *
171    * @throws PlanNaamInUseException when another planregistratie has the same plan name
172    */
173   @Transactional
174   public void insertPlanregistratie(Planregistratie planregistratie, String username) throws ParseException {
175     final UUID id = UUID.fromString(planregistratie.getId());
176     final OffsetDateTime now = OffsetDateTime.now(ZoneId.of("Europe/Amsterdam"));
177 
178     boolean planNaamInUse = this.jdbcClient
179         .sql("select exists(select 1 from planregistratie where plan_naam = ? and id <> ?)")
180         .param(planregistratie.getPlanNaam())
181         .param(id)
182         .query(Boolean.class)
183         .single();
184     if (planNaamInUse) {
185       throw new PlanNaamInUseException(planregistratie.getPlanNaam());
186     }
187 
188     Optional<Created> existing = this.jdbcClient
189         .sql("select creator, created_at from planregistratie where id = ?")
190         .param(id)
191         .query(Created.class)
192         .optional();
193     planregistratie
194         .setCreator(existing.map(Created::creator).orElse(username))
195         .setCreatedAt(existing.map(Created::createdAt).orElse(now))
196         .setEditor(existing.isPresent() ? username : null)
197         .setEditedAt(existing.isPresent() ? now : null);
198     this.deletePlanregistratie(planregistratie.getId());
199     try {
200       insertPlanregistratieRow(planregistratie);
201     } catch (DuplicateKeyException e) {
202       // Another planregistratie with the same name was inserted concurrently after the check above
203       if (e.getMessage() != null && e.getMessage().contains("planregistratie_plan_naam_key")) {
204         throw new PlanNaamInUseException(planregistratie.getPlanNaam());
205       }
206       throw e;
207     }
208 
209     for (NewPlancategorie p : AantallenRows.toPlancategorieen(planregistratie.getAantallen())) {
210       UUID plancategorieId = UUID.randomUUID();
211       this.jdbcClient
212           .sql("""
213 insert into plancategorie(id, planregistratie_id, %s, totaal_gepland, totaal_gerealiseerd)
214 values (:id, :planregistratieId, :value, :gepland, :gerealiseerd)""".formatted(p.column()))
215           .param("id", plancategorieId)
216           .param("planregistratieId", id)
217           .param("value", p.value(), Types.OTHER)
218           .param("gepland", p.aantal().gepland())
219           .param("gerealiseerd", p.aantal().gerealiseerd())
220           .update();
221       if (p.aantal().perJaar() != null) {
222         for (Map.Entry<Integer, Integer> jaar : p.aantal().perJaar().entrySet()) {
223           this.jdbcClient
224               .sql(
225                   "insert into detailplanning(plancategorie_id, jaartal, aantal_gepland) values (?, ?, ?)")
226               .param(plancategorieId)
227               .param(jaar.getKey())
228               .param(jaar.getValue())
229               .update();
230         }
231       }
232     }
233   }

235   private void insertPlanregistratieRow(Planregistratie planregistratie) throws ParseException {
236     this.jdbcClient
237         .sql("""
238 insert into planregistratie(
239 id,
240 geometrie,
241 creator,
242 created_at,
243 editor,
244 edited_at,
245 plan_naam,
246 provincie,
247 gemeente,
248 regio,
249 plaatsnaam,
250 vertrouwelijkheid,
251 opdrachtgever_type,
252 opdrachtgever_naam,
253 opmerkingen,
254 plantype,
255 bestemmingsplan,
256 status_project,
257 status_planologisch,
258 knelpunten_meerkeuze,
259 beoogd_woonmilieu_abf13,
260 aantal_studentenwoningen,
261 sleutelproject
262 )
263 values (
264 :id,
265 :geometrie,
266 :creator,
267 :createdAt,
268 :editor,
269 :editedAt,
270 :planNaam,
271 :provincie,
272 :gemeente,
273 :regio,
274 :plaatsnaam,
275 :vertrouwelijkheid,
276 :opdrachtgeverType,
277 :opdrachtgeverNaam,
278 :opmerkingen,
279 :plantype,
280 :bestemmingsplan,
281 :statusProject,
282 :statusPlanologisch,
283 cast(:knelpuntenMeerkeuze as pmw_knelpunten_meerkeuze[]),
284 :beoogdWoonmilieuAbf13,
285 :aantalStudentenwoningen,
286 :sleutelproject
287 )""")
288         .param("id", UUID.fromString(planregistratie.getId()))
289         .param("geometrie", wktToWkb(planregistratie.getGeometrie()))
290         .param("creator", planregistratie.getCreator())
291         .param("createdAt", planregistratie.getCreatedAt())
292         .param("editor", planregistratie.getEditor())
293         .param("editedAt", planregistratie.getEditedAt())
294         .param("planNaam", planregistratie.getPlanNaam())
295         .param("provincie", planregistratie.getProvincie())
296         .param("gemeente", planregistratie.getGemeente())
297         .param("regio", planregistratie.getRegio())
298         .param("plaatsnaam", planregistratie.getPlaatsnaam())
299         .param("vertrouwelijkheid", planregistratie.getVertrouwelijkheid(), Types.OTHER)
300         .param("opdrachtgeverType", planregistratie.getOpdrachtgeverType(), Types.OTHER)
301         .param("opdrachtgeverNaam", planregistratie.getOpdrachtgeverNaam())
302         .param("opmerkingen", planregistratie.getOpmerkingen())
303         .param("plantype", planregistratie.getPlantype(), Types.OTHER)
304         .param("bestemmingsplan", planregistratie.getBestemmingsplan())
305         .param("statusProject", planregistratie.getStatusProject(), Types.OTHER)
306         .param("statusPlanologisch", planregistratie.getStatusPlanologisch(), Types.OTHER)
307         .param("knelpuntenMeerkeuze", planregistratie.getKnelpuntenMeerkeuze(), Types.ARRAY)
308         .param("beoogdWoonmilieuAbf13", planregistratie.getBeoogdWoonmilieuAbf13(), Types.OTHER)
309         .param("aantalStudentenwoningen", planregistratie.getAantalStudentenwoningen())
310         .param("sleutelproject", planregistratie.getSleutelproject())
311         .update();
312   }

314   /** Returns the values of the enum types of the planmonitor wonen schema. */
315   public EnumValues getEnumValues() {
316     record EnumValue(String type, String value) {}
317     return new EnumValues(this.jdbcClient.sql("""
318 select t.typname as type, e.enumlabel as value
319 from pg_type t
320 join pg_enum e on e.enumtypid = t.oid
321 where t.typname like 'pmw\\_%' and pg_type_is_visible(t.oid)
322 order by t.typname, e.enumsortorder""").query(EnumValue.class).stream()
323         .collect(groupingBy(EnumValue::type, mapping(EnumValue::value, toList()))));
324   }

334   /** Returns whether the geometry intersects the geometry of the gemeente, or empty for an unknown gemeente. */
335   public Optional<Boolean> geometryIntersectsGemeente(String gemeente, String wkt) throws ParseException {
336     return this.jdbcClient
337         .sql("select ST_Intersects(geometry, ST_GeomFromWKB(?, 28992)) from gemeente where naam = ?")
338         .param(wktToWkb(wkt))
339         .param(gemeente)
340         .query(Boolean.class)
341         .optional();
342   }
