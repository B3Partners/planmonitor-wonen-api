51    public PlanmonitorWonenDatabaseService(JdbcClient jdbcClient) {
52      this.jdbcClient = jdbcClient;
53  
54      final GenericConversionService conversionService = new GenericConversionService();
55      DefaultConversionService.addDefaultConverters(conversionService);
56      conversionService.addConverter(new Converter<PgArray, String[]>() {
57        @Override
58        public String[] convert(PgArray source) {
59          try {
60            return (String[]) source.getArray();
61          } catch (SQLException e) {
62            throw new RuntimeException(e);
63          }
64        }
65      });
66  
67      planregistratieRowMapper = new SimplePropertyRowMapper<>(Planregistratie.class, conversionService) {
68        @Override
69        @NonNull public Planregistratie mapRow(@NonNull ResultSet rs, int rowNumber) throws SQLException {
70          Planregistratie planregistratie = super.mapRow(rs, rowNumber);
71          planregistratie.setGeometrie(wkbToWkt(planregistratie.getGeometrie()));
72          return planregistratie;
73        }
74      };
75    }
76  
77    private String wkbToWkt(String wkb) {
78      if (wkb == null) {
79        return null;
80      }
81      try {
82        Geometry geometry = new WKBReader().read(WKBReader.hexToBytes(wkb));
83        return new WKTWriter().write(geometry);
84      } catch (ParseException e) {
85        throw new RuntimeException(e);
86      }
87    }
88  
89    private byte[] wktToWkb(String wkt) throws ParseException {
90      Geometry geometry = new WKTReader().read(wkt);
91      return new WKBWriter().write(geometry);
92    }

94    private String sqlQuestionMarks(int count) {
95      String[] qs = new String[count];
96      Arrays.fill(qs, "?");
97      return String.join(", ", qs);
98    }

121   /**
122    * @param planregistratieFilter SQL condition for the planregistratie table, only use with constant strings and use
123    *     parameters for values
124    */
125   private List<Planregistratie> getPlanregistraties(
126       String planregistratieFilter, Map<String, ?> params, boolean withAantallen) {
127     List<Planregistratie> planregistraties = jdbcClient
128         .sql("select * from planregistratie where %s order by plan_naam".formatted(planregistratieFilter))
129         .params(params)
130         .query(planregistratieRowMapper)
131         .list();
132     if (!withAantallen || planregistraties.isEmpty()) {
133       return planregistraties;
134     }
135 
136     String planregistratieIds = "select id from planregistratie where " + planregistratieFilter;
137     Map<UUID, List<DetailplanningRow>> detailplanningen =
138         jdbcClient.sql("""
139 select d.plancategorie_id, d.jaartal, d.aantal_gepland
140 from detailplanning d
141 join plancategorie c on c.id = d.plancategorie_id
142 where c.planregistratie_id in (%s)""".formatted(planregistratieIds)).params(params).query(DetailplanningRow.class).stream()
143             .collect(groupingBy(DetailplanningRow::plancategorieId));
144 
145     // Ordered by the categorie values in enum order
146     Map<UUID, List<PlancategorieRow>> plancategorieen =
147         jdbcClient.sql("""
148 select * from plancategorie
149 where planregistratie_id in (%s)
150 order by nieuwbouw, woning_type, wonen_en_zorg, flexwoningen, betaalbaarheid, sloop""".formatted(planregistratieIds)).params(params).query(PlancategorieRow.class).stream()
151             .collect(groupingBy(PlancategorieRow::planregistratieId));
152 
153     planregistraties.forEach(p -> p.setAantallen(AantallenRows.toAantallen(
154         plancategorieen.getOrDefault(UUID.fromString(p.getId()), List.of()), detailplanningen)));
155     return planregistraties;
156   }

168   /**
169    * Inserts a planregistratie or replaces an existing one, including the aantallen. The creator, createdAt, editor
170    * and editedAt fields of the planregistratie are set from the username and the current time, the values from the
171    * client are ignored. For an existing planregistratie the creator and createdAt are kept.
172    */
173   @Transactional
174   public void insertPlanregistratie(Planregistratie planregistratie, String username) throws ParseException {
175     final String id = planregistratie.getId();
176     final OffsetDateTime now = OffsetDateTime.now(ZoneId.of("Europe/Amsterdam"));
177 
178     Optional<Created> existing = this.jdbcClient
179         .sql("select creator, created_at from planregistratie where id = ?")
180         .param(1, id, Types.OTHER)
181         .query(Created.class)
182         .optional();
183     planregistratie
184         .setCreator(existing.map(Created::creator).orElse(username))
185         .setCreatedAt(existing.map(Created::createdAt).orElse(now))
186         .setEditor(existing.isPresent() ? username : null)
187         .setEditedAt(existing.isPresent() ? now : null);
188     this.deletePlanregistratie(id);
189     String insertPlanregistratie = """
190 insert into planregistratie(
191 id,
192 geometrie,
193 creator,
194 created_at,
195 editor,
196 edited_at,
197 plan_naam,
198 provincie,
199 gemeente,
200 regio,
201 plaatsnaam,
202 vertrouwelijkheid,
203 opdrachtgever_type,
204 opdrachtgever_naam,
205 opmerkingen,
206 plantype,
207 bestemmingsplan,
208 status_project,
209 status_planologisch,
210 knelpunten_meerkeuze,
211 beoogd_woonmilieu_abf13,
212 aantal_studentenwoningen,
213 sleutelproject
214 )
215 values (%s, ?::pmw_knelpunten_meerkeuze[], %s)""".formatted(sqlQuestionMarks(19), sqlQuestionMarks(3));
216     this.jdbcClient
217         .sql(insertPlanregistratie)
218         .param(1, planregistratie.getId(), Types.OTHER)
219         .param(wktToWkb(planregistratie.getGeometrie()))
220         .param(planregistratie.getCreator())
221         .param(planregistratie.getCreatedAt())
222         .param(planregistratie.getEditor())
223         .param(planregistratie.getEditedAt())
224         .param(planregistratie.getPlanNaam())
225         .param(planregistratie.getProvincie())
226         .param(planregistratie.getGemeente())
227         .param(planregistratie.getRegio())
228         .param(planregistratie.getPlaatsnaam())
229         .param(12, planregistratie.getVertrouwelijkheid(), Types.OTHER)
230         .param(13, planregistratie.getOpdrachtgeverType(), Types.OTHER)
231         .param(planregistratie.getOpdrachtgeverNaam())
232         .param(planregistratie.getOpmerkingen())
233         .param(16, planregistratie.getPlantype(), Types.OTHER)
234         .param(planregistratie.getBestemmingsplan())
235         .param(18, planregistratie.getStatusProject(), Types.OTHER)
236         .param(19, planregistratie.getStatusPlanologisch(), Types.OTHER)
237         .param(20, planregistratie.getKnelpuntenMeerkeuze(), Types.ARRAY)
238         .param(21, planregistratie.getBeoogdWoonmilieuAbf13(), Types.OTHER)
239         .param(planregistratie.getAantalStudentenwoningen())
240         .param(planregistratie.getSleutelproject())
241         .update();
242 
243     for (NewPlancategorie p : AantallenRows.toPlancategorieen(planregistratie.getAantallen())) {
244       UUID plancategorieId = UUID.randomUUID();
245       this.jdbcClient
246           .sql("""
247 insert into plancategorie(id, planregistratie_id, %s, totaal_gepland, totaal_gerealiseerd)
248 values (?, ?, ?::pmw_%s, ?, ?)""".formatted(p.column(), p.column()))
249           .param(1, plancategorieId, Types.OTHER)
250           .param(2, id, Types.OTHER)
251           .param(3, p.value())
252           .param(4, p.aantal().gepland())
253           .param(5, p.aantal().gerealiseerd())
254           .update();
255       if (p.aantal().perJaar() != null) {
256         for (Map.Entry<Integer, Integer> jaar : p.aantal().perJaar().entrySet()) {
257           this.jdbcClient
258               .sql(
259                   "insert into detailplanning(plancategorie_id, jaartal, aantal_gepland) values (?, ?, ?)")
260               .param(1, plancategorieId, Types.OTHER)
261               .param(2, jaar.getKey())
262               .param(3, jaar.getValue())
263               .update();
264         }
265       }
266     }
267   }
