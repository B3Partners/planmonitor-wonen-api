9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.util.List;
12  import java.util.Map;
13  import java.util.TreeMap;
14  import java.util.UUID;
15  import nl.b3p.planmonitorwonen.api.AantallenRows.DetailplanningRow;
16  import nl.b3p.planmonitorwonen.api.AantallenRows.NewPlancategorie;
17  import nl.b3p.planmonitorwonen.api.AantallenRows.PlancategorieRow;
18  import nl.b3p.planmonitorwonen.api.model.Aantal;
19  import nl.b3p.planmonitorwonen.api.model.Aantallen;
20  import org.junit.jupiter.api.Test;
21  
22  class AantallenRowsTest {
23    private static final UUID PLAN = UUID.randomUUID();
24  
25    @Test
26    void rowsToAantallen() {
27      UUID nieuwbouw = UUID.randomUUID();
28      UUID eengezins = UUID.randomUUID();
29      List<PlancategorieRow> rows = List.of(
30          new PlancategorieRow(nieuwbouw, PLAN, "Nieuwbouw", null, null, null, null, null, 10, 1),
31          new PlancategorieRow(eengezins, PLAN, null, "Eengezins", null, null, null, null, 10, 1),
32          new PlancategorieRow(UUID.randomUUID(), PLAN, null, null, null, null, null, "Sloop", 2, 0));
33      Map<UUID, List<DetailplanningRow>> detailplanningen = Map.of(
34          nieuwbouw,
35          List.of(new DetailplanningRow(nieuwbouw, 2026, 5), new DetailplanningRow(nieuwbouw, 2025, 4)),
36          eengezins,
37          List.of(new DetailplanningRow(eengezins, 2025, 9)));
38  
39      Aantallen aantallen = AantallenRows.toAantallen(rows, detailplanningen);
40  
41      assertEquals(new Aantal(10, 1, new TreeMap<>(Map.of(2025, 4, 2026, 5))), aantallen.bruto());
42      assertEquals(
43          List.of(2025, 2026), List.copyOf(aantallen.bruto().perJaar().keySet()));
44      assertEquals(new Aantal(2, 0, new TreeMap<>()), aantallen.sloop());
45      assertEquals(Map.of("Eengezins", new Aantal(10, 1, new TreeMap<>(Map.of(2025, 9)))), aantallen.woningType());
46      assertEquals(Map.of(), aantallen.betaalbaarheid());
47    }
48  
49    @Test
50    void aantallenToPlancategorieen() {
51      Aantal aantal = new Aantal(10, 0, new TreeMap<>());
52      Aantallen aantallen = new Aantallen(
53          aantal, null, Map.of("Meergezins", aantal), null, Map.of(), Map.of("Sociale huur", aantal));
54  
55      assertEquals(
56          List.of(
57              new NewPlancategorie("nieuwbouw", "Nieuwbouw", aantal),
58              new NewPlancategorie("woning_type", "Meergezins", aantal),
59              new NewPlancategorie("betaalbaarheid", "Sociale huur", aantal)),
60          AantallenRows.toPlancategorieen(aantallen));
61      assertEquals(List.of(), AantallenRows.toPlancategorieen(null));
62    }
63  }
