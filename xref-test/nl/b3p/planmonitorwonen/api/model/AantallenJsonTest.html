7   package nl.b3p.planmonitorwonen.api.model;
8   
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.util.Map;
12  import java.util.TreeMap;
13  import org.junit.jupiter.api.Test;
14  import tools.jackson.databind.json.JsonMapper;
15  
16  class AantallenJsonTest {
17    private final JsonMapper mapper = JsonMapper.builder().build();
18  
19    @Test
20    void nullMeansNotFilledInAndZeroIsKept() {
21      Aantallen aantallen = mapper.readValue("""
22  {"bruto": {"gepland": 10, "perJaar": {"2025": 0, "2026": null, "2027": 10}},
23  "sloop": null,
24  "woningType": {"Eengezins": {"gepland": null, "gerealiseerd": null}}}""", Aantallen.class);
25  
26      assertEquals(new Aantal(10, 0, new TreeMap<>(Map.of(2025, 0, 2027, 10))), aantallen.bruto());
27      assertEquals(null, aantallen.sloop());
28      assertEquals(Map.of("Eengezins", new Aantal(0, 0, null)), aantallen.woningType());
29    }
30  
31    @Test
32    void categoriesNotFilledInAreLeftOut() {
33      Aantallen aantallen = new Aantallen(
34          new Aantal(10, 0, new TreeMap<>(Map.of(2027, 10))),
35          null,
36          Map.of("Meergezins", new Aantal(0, 0, null)),
37          Map.of(),
38          null,
39          Map.of());
40  
41      assertEquals(
42          "{\"bruto\":{\"gepland\":10,\"gerealiseerd\":0,\"perJaar\":{\"2027\":10}},"
43              + "\"woningType\":{\"Meergezins\":{\"gepland\":0,\"gerealiseerd\":0,\"perJaar\":{}}}}",
44          mapper.writeValueAsString(aantallen));
45    }
46  }
