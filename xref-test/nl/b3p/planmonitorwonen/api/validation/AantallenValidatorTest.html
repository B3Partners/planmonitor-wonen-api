7   package nl.b3p.planmonitorwonen.api.validation;

9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.util.ArrayList;
12  import java.util.List;
13  import java.util.Map;
14  import java.util.TreeMap;
15  import nl.b3p.planmonitorwonen.api.model.Aantal;
16  import nl.b3p.planmonitorwonen.api.model.Aantallen;
17  import org.junit.jupiter.api.Test;
18  
19  class AantallenValidatorTest {
20    private static Aantal aantal(int gepland, int gerealiseerd, Map<Integer, Integer> perJaar) {
21      return new Aantal(gepland, gerealiseerd, new TreeMap<>(perJaar));
22    }
23  
24    /** 20 dwellings, 5 realised, 10 in 2026 and 5 in 2027, divided over all groups. */
25    private static Aantallen valid() {
26      return new Aantallen(
27          aantal(20, 5, Map.of(2026, 10, 2027, 5)),
28          aantal(3, 0, Map.of()),
29          Map.of(
30              "Eengezins", aantal(12, 5, Map.of(2026, 7)),
31              "Meergezins", aantal(8, 0, Map.of(2026, 3, 2027, 5)),
32              "Onbekend", aantal(0, 0, Map.of(2026, 0))),
33          Map.of("Regulier", aantal(20, 5, Map.of(2026, 10, 2027, 5))),
34          Map.of("Regulier permanent", aantal(20, 5, Map.of(2026, 10, 2027, 5))),
35          Map.of(
36              "Sociale huur", aantal(10, 0, Map.of(2026, 5, 2027, 5)),
37              "Koop dure koop", aantal(10, 5, Map.of(2026, 5))));
38    }
39  
40    private static List<ValidationError> validate(Aantallen aantallen) {
41      List<ValidationError> errors = new ArrayList<>();
42      AantallenValidator.validate(aantallen, TestEnumValues.ENUM_VALUES, errors);
43      return errors;
44    }
45  
46    private static List<String> fields(List<ValidationError> errors) {
47      return errors.stream().map(ValidationError::field).toList();
48    }
49  
50    @Test
51    void validAantallen() {
52      assertEquals(List.of(), validate(valid()));
53    }
54  
55    @Test
56    void notFilledIn() {
57      assertEquals(List.of(), validate(null));
58      assertEquals(List.of(), validate(new Aantallen(null, null, null, null, null, null)));
59    }
60  
61    @Test
62    void sloopOnlyDoesNotNeedYears() {
63      assertEquals(List.of(), validate(new Aantallen(null, aantal(5, 0, Map.of()), null, null, null, null)));
64    }
65  
66    @Test
67    void everyGroupMustAddUpToBruto() {
68      Aantallen brutoOnly = new Aantallen(aantal(20, 0, Map.of(2027, 20)), null, null, null, null, null);
69      assertEquals(
70          List.of(
71              "aantallen.woningType",
72              "aantallen.woningType.perJaar.2027",
73              "aantallen.wonenEnZorg",
74              "aantallen.wonenEnZorg.perJaar.2027",
75              "aantallen.flexwoningen",
76              "aantallen.flexwoningen.perJaar.2027",
77              "aantallen.betaalbaarheid",
78              "aantallen.betaalbaarheid.perJaar.2027"),
79          fields(validate(brutoOnly)));
80      assertEquals(
81          "Totaal gepland van woningType (0) is niet gelijk aan bruto (20)",
82          validate(brutoOnly).getFirst().message());
83    }
84  
85    @Test
86    void groupWithoutBruto() {
87      Aantallen aantallen =
88          new Aantallen(null, null, Map.of("Eengezins", aantal(5, 0, Map.of(2026, 5))), null, null, null);
89      assertEquals(List.of("aantallen.woningType", "aantallen.woningType.perJaar.2026"), fields(validate(aantallen)));
90    }
91  
92    @Test
93    void yearsMustAddUpToRestcapaciteit() {
94      Aantallen v = valid();
95      Aantallen aantallen = new Aantallen(
96          v.bruto(),
97          v.sloop(),
98          Map.of(
99              // 2027 moved to 2028, so the year totals of woningType do not add up either
100             "Eengezins", aantal(12, 5, Map.of(2026, 7)),
101             "Meergezins", aantal(8, 0, Map.of(2026, 3, 2028, 4))),
102         v.wonenEnZorg(),
103         v.flexwoningen(),
104         v.betaalbaarheid());
105     List<ValidationError> errors = validate(aantallen);
106     assertEquals(
107         List.of(
108             "aantallen.woningType.Meergezins.perJaar",
109             "aantallen.woningType.perJaar.2027",
110             "aantallen.woningType.perJaar.2028"),
111         fields(errors));
112     assertEquals(
113         "Totaal per jaar (7) is niet gelijk aan gepland min gerealiseerd (8)",
114         errors.getFirst().message());
115   }
116 
117   @Test
118   void invalidValuesSkipTotals() {
119     Aantallen v = valid();
120     Aantallen aantallen = new Aantallen(
121         v.bruto(),
122         aantal(-1, 0, Map.of()),
123         Map.of(
124             "Tussenwoning", aantal(12, 5, Map.of(2026, 7)),
125             "Meergezins", aantal(8, 9, Map.of(2050, 3, 2027, -5))),
126         v.wonenEnZorg(),
127         v.flexwoningen(),
128         v.betaalbaarheid());
129     List<ValidationError> errors = validate(aantallen);
130     assertEquals(
131         List.of(
132             "aantallen.sloop.gepland",
133             "aantallen.woningType.Meergezins.gerealiseerd",
134             "aantallen.woningType.Meergezins.perJaar.2027",
135             "aantallen.woningType.Meergezins.perJaar.2050",
136             "aantallen.woningType.Tussenwoning"),
137         fields(errors).stream().sorted().toList());
138     assertEquals(
139         "Onbekende categorie \"Tussenwoning\", mogelijke waarden: Eengezins, Meergezins, Onbekend",
140         errors.stream()
141             .filter(e -> e.field().equals("aantallen.woningType.Tussenwoning"))
142             .findFirst()
143             .orElseThrow()
144             .message());
145   }
