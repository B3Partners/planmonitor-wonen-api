7   package nl.b3p.planmonitorwonen.api.validation;
8   
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.util.List;
12  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
13  import org.junit.jupiter.api.Test;
14  
15  class PlanregistratieValidatorTest {
16    private static Planregistratie minimal() {
17      return new Planregistratie()
18          .setPlanNaam("Plan")
19          .setGemeente("Middelburg")
20          .setPlantype("Herstructurering")
21          .setStatusPlanologisch("3. In voorbereiding")
22          .setGeometrie("POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000, 30000 390000))");
23    }
24  
25    @Test
26    void provincieSpecificFieldsAreOptional() {
27      assertEquals(List.of(), PlanregistratieValidator.validate(minimal()));
28    }
29  
30    @Test
31    void reportsAllMissingRequiredFields() {
32      List<ValidationError> errors = PlanregistratieValidator.validate(new Planregistratie().setPlanNaam(" "));
33      assertEquals(
34          List.of("planNaam", "gemeente", "plantype", "statusPlanologisch", "geometrie"),
35          errors.stream().map(ValidationError::field).toList());
36    }
37  
38    @Test
39    void geometrieMustBeValidPolygon() {
40      assertEquals(
41          List.of(new ValidationError("geometrie", "Geometrie moet een polygoon zijn, geen Point")),
42          PlanregistratieValidator.validate(minimal().setGeometrie("POINT (30000 390000)")));
43      assertEquals(
44          "geometrie",
45          PlanregistratieValidator.validate(minimal().setGeometrie("POLYGON ((30000"))
46              .getFirst()
47              .field());
48    }
49  }
