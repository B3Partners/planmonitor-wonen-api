7   package nl.b3p.planmonitorwonen.api.validation;
8   
9   import static org.junit.jupiter.api.Assertions.assertEquals;
10  
11  import java.util.List;
12  import nl.b3p.planmonitorwonen.api.model.Planregistratie;
13  import org.junit.jupiter.api.Test;
14  
15  class PlanregistratieValidatorTest {
16    private static List<ValidationError> validate(Planregistratie planregistratie) {
17      return PlanregistratieValidator.validate(planregistratie, TestEnumValues.ENUM_VALUES);
18    }
19  
20    private static Planregistratie minimal() {
21      return new Planregistratie()
22          .setPlanNaam("Plan")
23          .setGemeente("Middelburg")
24          .setPlantype("Herstructurering")
25          .setStatusPlanologisch("3. In voorbereiding")
26          .setGeometrie("POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000, 30000 390000))");
27    }
28  
29    @Test
30    void provincieSpecificFieldsAreOptional() {
31      assertEquals(List.of(), validate(minimal()));
32    }
33  
34    @Test
35    void reportsAllMissingRequiredFields() {
36      List<ValidationError> errors = validate(new Planregistratie().setPlanNaam(" "));
37      assertEquals(
38          List.of("planNaam", "gemeente", "plantype", "statusPlanologisch", "geometrie"),
39          errors.stream().map(ValidationError::field).toList());
40    }
41  
42    @Test
43    void geometrieMustBeValidPolygon() {
44      assertEquals(
45          List.of(new ValidationError("geometrie", "Geometrie moet een polygoon zijn, geen Point")),
46          validate(minimal().setGeometrie("POINT (30000 390000)")));
47      assertEquals(
48          "geometrie",
49          validate(minimal().setGeometrie("POLYGON ((30000")).getFirst().field());
50    }
51  
52    @Test
53    void enumFieldsMustHaveKnownValue() {
54      List<ValidationError> errors = validate(minimal()
55          .setPlantype("Onbekend")
56          .setStatusPlanologisch("3")
57          .setVertrouwelijkheid("Geheim")
58          .setOpdrachtgeverType("Iemand")
59          .setStatusProject("Klaar")
60          .setBeoogdWoonmilieuAbf13("Stad")
61          .setKnelpuntenMeerkeuze(new String[] {"Stikstof", "Netcongestie"}));
62      assertEquals(
63          List.of(
64              "plantype",
65              "statusPlanologisch",
66              "vertrouwelijkheid",
67              "opdrachtgeverType",
68              "statusProject",
69              "beoogdWoonmilieuAbf13",
70              "knelpuntenMeerkeuze[1]"),
71          errors.stream().map(ValidationError::field).toList());
72      assertEquals(
73          "Onbekende waarde \"Geheim\", mogelijke waarden: Gemeente, Openbaar, Provincie",
74          errors.get(2).message());
75    }
76  
77    @Test
78    void optionalFieldsWithValidValues() {
79      assertEquals(
80          List.of(),
81          validate(minimal()
82              .setVertrouwelijkheid("Gemeente")
83              .setOpdrachtgeverType("Woningbouwcorporatie")
84              .setStatusProject("Realisatie tot eerste woning")
85              .setBeoogdWoonmilieuAbf13("Geen Woonmilieu")
86              .setKnelpuntenMeerkeuze(new String[] {"Grond/Didam", "Nuts", "Spuitzones/milieu"})
87              .setAantalStudentenwoningen(0)));
88    }
89  
90    @Test
91    void knelpuntMustNotBeNull() {
92      assertEquals(
93          List.of(new ValidationError("knelpuntenMeerkeuze[0]", "Knelpunt mag niet leeg zijn")),
94          validate(minimal().setKnelpuntenMeerkeuze(new String[] {null})));
95    }
96  
97    @Test
98    void planNaamMinimumLength() {
99      assertEquals(
100         List.of(new ValidationError("planNaam", "Plannaam moet minimaal 3 tekens lang zijn")),
101         validate(minimal().setPlanNaam(" ab ")));
102     assertEquals(List.of(), validate(minimal().setPlanNaam("abc")));
103   }
104 
105   @Test
106   void aantalStudentenwoningenNotNegative() {
107     assertEquals(
108         List.of(new ValidationError(
109             "aantalStudentenwoningen", "Aantal studentenwoningen mag niet negatief zijn")),
110         validate(minimal().setAantalStudentenwoningen(-1)));
111   }
112 
113   @Test
114   void geometrieMustBeInRd() {
115     List<ValidationError> errors =
116         validate(minimal().setGeometrie("POLYGON ((3.6 51.5, 3.7 51.5, 3.7 51.6, 3.6 51.6, 3.6 51.5))"));
117     assertEquals(
118         List.of(new ValidationError(
119             "geometrie",
120             "Geometrie ligt niet in Nederland, de coördinaten moeten in RD (EPSG:28992) zijn")),
121         errors);
122   }
123 
124   @Test
125   void geometrieMustBeTopologicallyValid() {
126     // Bow tie
127     List<ValidationError> errors = validate(minimal()
128         .setGeometrie("POLYGON ((30000 390000, 31000 391000, 31000 390000, 30000 391000, 30000 390000))"));
129     assertEquals(
130         List.of(new ValidationError(
131             "geometrie",
132             "Geometrie is ongeldig: Self-intersection at or near point (30500.0, 390500.0, NaN)")),
133         errors);
134   }
135 
136   @Test
137   void geometrieRingNotClosed() {
138     List<ValidationError> errors =
139         validate(minimal().setGeometrie("POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000))"));
140     assertEquals(
141         List.of("geometrie"),
142         errors.stream().map(ValidationError::field).toList());
143   }
144 
145   @Test
146   void geometrieNotEmpty() {
147     assertEquals(
148         List.of(new ValidationError("geometrie", "Geometrie is leeg")),
149         validate(minimal().setGeometrie("POLYGON EMPTY")));
150   }
151 }
