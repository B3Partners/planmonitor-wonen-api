39  /**
40   * CSRF protection is disabled in {@link ApiSecurityConfig}. These tests check the assumptions that make that safe, so
41   * that a change that breaks them fails here instead of silently allowing CSRF.
42   */
43  @WebMvcTest(
44      controllers = {
45        GemeentesController.class,
46        PlanregistratieAutofillController.class,
47        PlanregistratieController.class,
48        UserInfoController.class
49      })
50  @Import(ApiSecurityConfig.class)
51  @ActiveProfiles("integration-test")
52  class CsrfSafetyTest {
53    /**
54     * POST endpoints that do not change any data. A cross-origin page can send a POST without a CORS preflight (with a
55     * form or text/plain content type), so an endpoint that changes data must use PUT, PATCH or DELETE instead, or CSRF
56     * protection must be enabled.
57     */
58    private static final Set<String> READ_ONLY_POST_PATTERNS =
59        Set.of("/{version}/planregistratie/autofill-by-geometry");
60  
61    @Value("${planmonitor-wonen-api.base-path}")
62    private String basePath;
63  
64    @Autowired
65    private WebApplicationContext context;
66  
67    @MockitoBean
68    private PlanmonitorAuthenticationService authenticationService;
69  
70    @MockitoBean
71    private PlanmonitorWonenDatabaseService planmonitorWonenDatabaseService;
72  
73    @MockitoBean
74    private PlanregistratieCompletionService completionService;
75  
76    @MockitoBean
77    private PlanregistratieAutofillService autofillService;
78  
79    @MockitoBean
80    private JdbcClient jdbcClient;
81  
82    @MockitoBean
83    private EnumValues enumValues;
84  
85    @Autowired
86    @Qualifier("requestMappingHandlerMapping") private RequestMappingHandlerMapping handlerMapping;
87  
88    private MockMvcTester mvc;
89  
90    @BeforeEach
91    void setUp() {
92      mvc = MockMvcTester.create(MockMvcBuilders.webAppContextSetup(context)
93          .apply(springSecurity())
94          .build());
95    }
96  
97    @Test
98    void controllersAreMapped() {
99      assertThat(handlerMapping.getHandlerMethods().values())
100         .extracting(handlerMethod -> handlerMethod.getBeanType().getSimpleName())
101         .contains("PlanregistratieController", "PlanregistratieAutofillController");
102   }
103 
104   @Test
105   void onlyReadOnlyPostEndpoints() {
106     Set<String> postPatterns = handlerMapping.getHandlerMethods().entrySet().stream()
107         .filter(entry -> entry.getValue().getBeanType().getPackageName().startsWith("nl.b3p.planmonitorwonen"))
108         .filter(entry -> {
109           Set<RequestMethod> methods =
110               entry.getKey().getMethodsCondition().getMethods();
111           // A mapping without methods accepts all methods, including POST
112           return methods.isEmpty() || methods.contains(RequestMethod.POST);
113         })
114         .flatMap(entry -> entry.getKey().getPatternValues().stream())
115         .map(pattern -> pattern.substring(basePath.length()))
116         .collect(Collectors.toSet());
117 
118     assertThat(postPatterns)
119         .as("POST endpoints must be read-only, see ApiSecurityConfig")
120         .isEqualTo(READ_ONLY_POST_PATTERNS);
121   }
122 
123   /**
124    * Without a CORS configuration Spring answers a preflight with 200 but without Access-Control-Allow-* headers, so
125    * the browser does not send the actual cross-origin request.
126    */
127   @Test
128   void crossOriginPreflightIsNotAllowed() {
129     assertThat(mvc.options()
130             .uri(basePath + "/v2/planregistratie/0b5c1d0e-3f43-4e4a-9a6e-2f1f6b6d2a10")
131             .header(HttpHeaders.ORIGIN, "https://attacker.example.com")
132             .header(HttpHeaders.ACCESS_CONTROL_REQUEST_METHOD, "PUT")
133             .header(HttpHeaders.ACCESS_CONTROL_REQUEST_HEADERS, "content-type"))
134         .doesNotContainHeader(HttpHeaders.ACCESS_CONTROL_ALLOW_ORIGIN)
135         .doesNotContainHeader(HttpHeaders.ACCESS_CONTROL_ALLOW_METHODS)
136         .doesNotContainHeader(HttpHeaders.ACCESS_CONTROL_ALLOW_CREDENTIALS);
137   }
