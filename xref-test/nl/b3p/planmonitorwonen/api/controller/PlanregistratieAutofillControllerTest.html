16  import nl.b3p.planmonitorwonen.api.security.PlanmonitorAuthenticationService;
17  import nl.b3p.planmonitorwonen.api.service.PlanregistratieAutofillService;
18  import org.junit.jupiter.api.Test;
19  import org.springframework.beans.factory.annotation.Autowired;
20  import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
21  import org.springframework.boot.webmvc.test.autoconfigure.WebMvcTest;
22  import org.springframework.http.HttpStatus;
23  import org.springframework.http.MediaType;
24  import org.springframework.jdbc.core.simple.JdbcClient;
25  import org.springframework.test.context.ActiveProfiles;
26  import org.springframework.test.context.bean.override.mockito.MockitoBean;
27  import org.springframework.test.web.servlet.assertj.MockMvcTester;
28  import org.springframework.web.server.ResponseStatusException;
29  
30  @WebMvcTest(
31      controllers = PlanregistratieAutofillController.class,
32      properties = {"planmonitor-wonen-api.base-path=/api/planmonitor-wonen", "spring.mvc.problemdetails.enabled=true"
33      })
34  @AutoConfigureMockMvc(addFilters = false)
35  @ActiveProfiles("integration-test")
36  class PlanregistratieAutofillControllerTest {
37    @Autowired
38    private MockMvcTester mvc;
39  
40    @MockitoBean
41    private PlanmonitorAuthenticationService authenticationService;
42  
43    @MockitoBean
44    private JdbcClient jdbcClient;
45  
46    @MockitoBean
47    private PlanregistratieAutofillService autofillService;
48  
49    @Test
50    void notLoggedInIsUnauthorized() {
51      when(authenticationService.requirePlanmonitorUser())
52          .thenThrow(new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Niet ingelogd"));
53  
54      assertThat(mvc.post()
55              .uri("/api/planmonitor-wonen/planregistratie/autofill-by-geometry")
56              .contentType(MediaType.TEXT_PLAIN)
57              .content("POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000, 30000 390000))"))
58          .hasStatus(HttpStatus.UNAUTHORIZED);
59      verifyNoInteractions(jdbcClient);
60      // The autofill service is also called by its startup event listener, so only check the lookups
61      verify(autofillService, never()).getProvincies(any());
62      verify(autofillService, never()).getRegios(any());
63      verify(autofillService, never()).getWoonmilieus(any());
64    }
65  }
