12  import static org.mockito.Mockito.doThrow;
13  import static org.mockito.Mockito.never;
14  import static org.mockito.Mockito.verify;
15  import static org.mockito.Mockito.when;

17  import java.util.Set;
18  import java.util.UUID;
19  import nl.b3p.planmonitorwonen.api.PlanmonitorWonenDatabaseService;
20  import nl.b3p.planmonitorwonen.api.security.PlanmonitorAuthenticationService;

24  import nl.b3p.planmonitorwonen.api.validation.PlanNaamInUseException;
25  import nl.b3p.planmonitorwonen.api.validation.TestEnumValues;
26  import org.junit.jupiter.api.Test;
27  import org.springframework.beans.factory.annotation.Autowired;
28  import org.springframework.boot.test.context.TestConfiguration;
29  import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
30  import org.springframework.boot.webmvc.test.autoconfigure.WebMvcTest;
31  import org.springframework.context.annotation.Bean;
32  import org.springframework.http.HttpStatus;
33  import org.springframework.http.MediaType;
34  import org.springframework.security.authentication.TestingAuthenticationToken;
35  import org.springframework.test.context.ActiveProfiles;
36  import org.springframework.test.context.bean.override.mockito.MockitoBean;
37  import org.springframework.test.web.servlet.assertj.MockMvcTester;
38  import org.springframework.web.server.ResponseStatusException;
39  
40  @WebMvcTest(
41      controllers = PlanregistratieController.class,
42      properties = {"planmonitor-wonen-api.base-path=/api/planmonitor-wonen", "spring.mvc.problemdetails.enabled=true"
43      })
44  @AutoConfigureMockMvc(addFilters = false)
45  @ActiveProfiles("integration-test")
46  class PlanregistratieControllerTest {
47    private static final String ID = "0b5c1d0e-3f43-4e4a-9a6e-2f1f6b6d2a10";
48    private static final String URL = "/api/planmonitor-wonen/planregistratie/" + ID;
49  
50    private static final String VALID_PLAN = """
51  {
52  "planNaam": "Plan",
53  "gemeente": "Middelburg",
54  "plantype": "Herstructurering",
55  "statusPlanologisch": "3. In voorbereiding",
56  "geometrie": "POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000, 30000 390000))"
57  }""";
58  
59    @TestConfiguration
60    static class EnumValuesConfiguration {
61      @Bean
62      EnumValues enumValues() {
63        return TestEnumValues.ENUM_VALUES;
64      }
65    }
66  
67    @Autowired
68    private MockMvcTester mvc;
69  
70    @MockitoBean
71    private PlanmonitorAuthenticationService authenticationService;
72  
73    @MockitoBean
74    private PlanmonitorWonenDatabaseService pmwDb;
75  
76    @MockitoBean
77    private PlanregistratieCompletionService completionService;
78  
79    private void loggedInAsGemeente() {
80      when(authenticationService.requirePlanmonitorUser())
81          .thenReturn(new PlanmonitorAuthentication(
82              new TestingAuthenticationToken("gemeente", null), false, Set.of("Middelburg")));
83    }
84  
85    @Test
86    void savesValidPlan() throws Exception {
87      loggedInAsGemeente();
88      assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(VALID_PLAN))
89          .hasStatusOk();
90      verify(pmwDb).insertPlanregistratie(any(), anyString());
91    }
92  
93    @Test
94    void notLoggedInIsUnauthorizedBeforeValidation() throws Exception {
95      when(authenticationService.requirePlanmonitorUser())
96          .thenThrow(new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Niet ingelogd"));
97      assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content("{}"))
98          .hasStatus(HttpStatus.UNAUTHORIZED);
99      verify(pmwDb, never()).insertPlanregistratie(any(), anyString());
100   }
101 
102   @Test
103   void invalidPlanIsRejectedWithErrors() throws Exception {
104     loggedInAsGemeente();
105     assertThat(mvc.put()
106             .uri(URL)
107             .contentType(MediaType.APPLICATION_JSON)
108             .content(VALID_PLAN.replace("Herstructurering", "Onbekend")))
109         .hasStatus(HttpStatus.BAD_REQUEST)
110         .bodyJson()
111         .extractingPath("$.errors[0].field")
112         .isEqualTo("plantype");
113     verify(pmwDb, never()).insertPlanregistratie(any(), anyString());
114   }
115 
116   @Test
117   void planNaamInUseIsConflict() throws Exception {
118     loggedInAsGemeente();
119     doThrow(new PlanNaamInUseException("Plan")).when(pmwDb).insertPlanregistratie(any(), anyString());
120     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(VALID_PLAN))
121         .hasStatus(HttpStatus.CONFLICT)
122         .bodyJson()
123         .extractingPath("$.detail")
124         .isEqualTo("Er bestaat al een andere planregistratie met de naam \"Plan\"");
125   }
126 
127   @Test
128   void idInBodyMustMatchPath() {
129     loggedInAsGemeente();
130     String body = VALID_PLAN.replace("{", "{\"id\": \"" + UUID.randomUUID() + "\",");
131     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(body))
132         .hasStatus(HttpStatus.BAD_REQUEST);
133   }
134 }
