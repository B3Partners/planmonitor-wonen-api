36  import org.springframework.context.annotation.Bean;
37  import org.springframework.http.HttpStatus;
38  import org.springframework.http.MediaType;
39  import org.springframework.security.authentication.TestingAuthenticationToken;
40  import org.springframework.test.context.ActiveProfiles;
41  import org.springframework.test.context.bean.override.mockito.MockitoBean;
42  import org.springframework.test.web.servlet.assertj.MockMvcTester;
43  import org.springframework.web.server.ResponseStatusException;

45  @WebMvcTest(
46      controllers = PlanregistratieController.class,
47      properties = {"planmonitor-wonen-api.base-path=/api/planmonitor-wonen", "spring.mvc.problemdetails.enabled=true"
48      })
49  @AutoConfigureMockMvc(addFilters = false)
50  @ActiveProfiles("integration-test")
51  @ExtendWith(OutputCaptureExtension.class)
52  class PlanregistratieControllerTest {
53    private static final String ID = "0b5c1d0e-3f43-4e4a-9a6e-2f1f6b6d2a10";
54    private static final String URL = "/api/planmonitor-wonen/v2/planregistratie/" + ID;
55  
56    private static final String VALID_PLAN = """
57  {
58  "planNaam": "Plan",
59  "gemeente": "Middelburg",
60  "plantype": "Herstructurering",
61  "statusPlanologisch": "3. In voorbereiding",
62  "geometrie": "POLYGON ((30000 390000, 31000 390000, 31000 391000, 30000 391000, 30000 390000))"
63  }""";
64  
65    @TestConfiguration
66    static class EnumValuesConfiguration {
67      @Bean
68      EnumValues enumValues() {
69        return TestEnumValues.ENUM_VALUES;
70      }
71    }
72  
73    @Autowired
74    private MockMvcTester mvc;
75  
76    @MockitoBean
77    private PlanmonitorAuthenticationService authenticationService;
78  
79    @MockitoBean
80    private PlanmonitorWonenDatabaseService pmwDb;
81  
82    @MockitoBean
83    private PlanregistratieCompletionService completionService;
84  
85    private void loggedInAsGemeente() {
86      when(authenticationService.requirePlanmonitorUser())
87          .thenReturn(new PlanmonitorAuthentication(
88              new TestingAuthenticationToken("gemeente", null), false, Set.of("Middelburg")));
89    }
90  
91    private void savedPlanIsReturned() {
92      when(pmwDb.getPlanregistratie(ID))
93          .thenReturn(Optional.of(new Planregistratie()
94              .setId(ID)
95              .setPlanNaam("Plan")
96              .setGemeente("Middelburg")
97              .setProvincie("Zeeland")));
98    }
99  
100   @Test
101   void createsNewPlan(CapturedOutput output) throws Exception {
102     loggedInAsGemeente();
103     savedPlanIsReturned();
104     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(VALID_PLAN))
105         .hasStatus(HttpStatus.CREATED)
106         .bodyJson()
107         .isLenientlyEqualTo("""
108 {"id": "%s", "planNaam": "Plan", "provincie": "Zeeland"}""".formatted(ID));
109     verify(pmwDb).insertPlanregistratie(any(), anyString());
110     assertThat(output)
111         .contains("User \"gemeente\" created plan id %s, name \"Plan\" of gemeente Middelburg".formatted(ID));
112   }
113 
114   @Test
115   void replacesExistingPlan(CapturedOutput output) throws Exception {
116     loggedInAsGemeente();
117     savedPlanIsReturned();
118     when(pmwDb.getPlanregistratieGemeente(ID)).thenReturn("Middelburg");
119     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(VALID_PLAN))
120         .hasStatusOk()
121         .bodyJson()
122         .extractingPath("$.provincie")
123         .isEqualTo("Zeeland");
124     verify(pmwDb).insertPlanregistratie(any(), anyString());
125     assertThat(output).contains("User \"gemeente\" replaced plan id " + ID);
126   }
127 
128   @Test
129   void notLoggedInIsUnauthorizedBeforeValidation() throws Exception {
130     when(authenticationService.requirePlanmonitorUser())
131         .thenThrow(new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Niet ingelogd"));
132     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content("{}"))
133         .hasStatus(HttpStatus.UNAUTHORIZED);
134     verify(pmwDb, never()).insertPlanregistratie(any(), anyString());
135   }
136 
137   @Test
138   void invalidPlanIsRejectedWithErrors() throws Exception {
139     loggedInAsGemeente();
140     assertThat(mvc.put()
141             .uri(URL)
142             .contentType(MediaType.APPLICATION_JSON)
143             .content(VALID_PLAN.replace("Herstructurering", "Onbekend")))
144         .hasStatus(HttpStatus.BAD_REQUEST)
145         .bodyJson()
146         .extractingPath("$.errors[0].field")
147         .isEqualTo("plantype");
148     verify(pmwDb, never()).insertPlanregistratie(any(), anyString());
149   }
150 
151   @Test
152   void planNaamInUseIsConflict() throws Exception {
153     loggedInAsGemeente();
154     doThrow(new PlanNaamInUseException("Plan")).when(pmwDb).insertPlanregistratie(any(), anyString());
155     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(VALID_PLAN))
156         .hasStatus(HttpStatus.CONFLICT)
157         .bodyJson()
158         .extractingPath("$.detail")
159         .isEqualTo("Er bestaat al een andere planregistratie met de naam \"Plan\"");
160   }
161 
162   @Test
163   void deletesPlan(CapturedOutput output) {
164     loggedInAsGemeente();
165     when(pmwDb.getPlanregistratieGemeente(ID)).thenReturn("Middelburg");
166     when(pmwDb.deletePlanregistratie(ID)).thenReturn(Optional.of("Plan"));
167     assertThat(mvc.delete().uri(URL)).hasStatusOk();
168     verify(pmwDb).deletePlanregistratie(ID);
169     assertThat(output)
170         .contains("User \"gemeente\" deleted plan id %s, name \"Plan\" of gemeente Middelburg".formatted(ID));
171   }
172 
173   @Test
174   void deleteUnknownPlanIsNotFound(CapturedOutput output) {
175     loggedInAsGemeente();
176     assertThat(mvc.delete().uri(URL)).hasStatus(HttpStatus.NOT_FOUND);
177     verify(pmwDb, never()).deletePlanregistratie(anyString());
178     assertThat(output).contains("User \"gemeente\" tried to delete unknown plan id " + ID);
179   }
180 
181   @Test
182   void idInBodyMustMatchPath() {
183     loggedInAsGemeente();
184     String body = VALID_PLAN.replace("{", "{\"id\": \"" + UUID.randomUUID() + "\",");
185     assertThat(mvc.put().uri(URL).contentType(MediaType.APPLICATION_JSON).content(body))
186         .hasStatus(HttpStatus.BAD_REQUEST);
187   }
